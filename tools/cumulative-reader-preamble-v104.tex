% \sphereplot{n} gives the plot codes for the nth sphere

\newcommand{\sphereplot}[1]{
  circle
    [radius=(#1)*\pgfkeysvalueof{/tikz/layerwidth}-
      \pgfkeysvalueof{/tikz/layerwidth}*\pgfkeysvalueof{/tikz/innerfactor}]
}


% \spheresystem[options]{n} draws a sphere system centered at
% the origin with n number of layers

\newcommand{\spheresystem}[2][]{
  \foreach \i in {1,...,#2}{
    \draw[sphere,#1] \sphereplot{\i} ;
  }
}

% \spherelayer[options]{n} shades the n-th layer

\newcommand{\spherelayer}[2][]{
  \begin{scope}[even odd rule]
    \fill[#1,sphere layer]
    \sphereplot{#2-1} \sphereplot{#2} ;
  \end{scope}
}

% \spherefill[options]{n} fills the nth sphere

\newcommand{\spherefill}[2][]{
    \fill[sphere intersection,#1]
    \sphereplot{#2} ;
}

% \sphereintersect[options]{path}{n} draws the path and shades the area
% of the path in the n th layer. Options only apply to the sphere layer

\newcommand{\sphereintersect}[3][]{
  \begin{scope}[even odd rule]
    \path[clip] #3;
    \spherefill[#1]{#2}
  \end{scope}
  \draw #3;
}

% \sphereproposition[options]{direction}{n}{width}{length} draws a
% proposition intersecting the n-th layer in direction away from the
% center of the sphere system, with endpoints length away from the
% vertex at an angle of width/2. Note that options applies to \draw,
% not to plot. To change the tension on the plot, e.g., say
% \tikzset{proposition/.style={smooth,tension=2}} -- in a group to
% make it apply only to a single proposition.

\newcommand{\propositionplot}[4]{
  plot [proposition]
  coordinates {+(#1+#3/2:#4)
    +(#1:#2*\pgfkeysvalueof{/tikz/layerwidth}-
    \pgfkeysvalueof{/tikz/layerwidth}*.9
    -\pgfkeysvalueof{/tikz/layerwidth}*\pgfkeysvalueof{/tikz/innerfactor})
    +(#1-#3/2:#4)}
}

\newcommand{\proposition}[5][]{
  \draw[proposition,#1] \propositionplot {#2}{#3}{#4}{#5} ;
  }

% \spherepropositionintersect does the same as \sphereproposition but
% also shades the are of intersection with the closest sphere

\newcommand{\propositionintersect}[5][]{
  \begin{scope}
  \path[clip] \propositionplot{#2}{#3}{#4}{#5};
  \spherefill[#1]{#3}
  \end{scope}
  \draw[proposition,#1] \propositionplot{#2}{#3}{#4}{#5};
}

% \spherepos[options]{direction}{n}{code} shifts the scope to a
% position in the center of the nth layer in direction angle from the
% center -- and then puts code there.

\newcommand{\spherepos}[4][]{
  \begin{scope}[#1,shift=(#2:#3*\pgfkeysvalueof{/tikz/layerwidth}-
      \pgfkeysvalueof{/tikz/layerwidth}/2-
        \pgfkeysvalueof{/tikz/layerwidth}*\pgfkeysvalueof{/tikz/innerfactor})]
    \path #4 ;
  \end{scope}
}

{\Huge خلاص منطق}\par\bigskip
{\Large پښتو — پاکستان}\par\bigskip
{\LARGE له بنسټونو څخه د سټونو تر تاريخ}\par\bigskip
د ماشيني ژباړې د روان کار ټوليزه نسخه
\end{center}
\bigskip
\small
په دې ټوليزه لوستيزه نسخه کښې له \LR{\latinfont OLP-0001} څخه تر
\LR{\latinfont OLP-0638} پورې د OLP_READER_UNITS سرچينيزو واحدونو متن د
منجمد اوپن لاجک د اصلي ټاکلي پروفايل له مخې وړاندې کېږي۔ دا د روان کار
نسخه ده؛ د ټولو ۷۲۲ واحدونو بشپړه ژباړه او د بشپړ کتاب کتنه لا روانه ده۔
له بنسټونو، سټونو، منطقي ژبو، ثبوتونو او محاسبويت څخه تر مودالي او
شهودي منطق، د سټونو ژورې تيورۍ، د ثبوت طريقو، استقرا او د منطقپوهانو
بشپړو ژوندليکونو پورې منل شوې مسودې رانغاړي۔ د بديلو او د پروفايل
د نورو انتخابونو بشپړې سمون وړ سرچينې ساتل کېږي؛ د وروستۍ بشپړې نسخې
د ټولو انتخابونو يوځای کول لا پاتې کار دے۔

تر \LR{\latinfont OLP-0311} پورې ژباړه او کتنه
\LR{\latinfont OpenAI Codex — GPT-5.6 Sol, Ultra} کړې ده۔
د \LR{\latinfont OLP-0312–0609} پخوانۍ مسودې او په
\LR{\latinfont OLP-0177} کښې د دوو نومونو سمون
\LR{\latinfont OpenAI Codex — GPT-6 Sol, Ultra} کړي دي۔
د \LR{\latinfont OLP-0610–0638} نوې ژباړه، د لوستونکي دا ترميمونه
او روانه خپلواکه بياکتنه \LR{\latinfont OpenAI Codex — GPT-6.1 Sol, Ultra}
کوي۔ د پخوانيو ۲۹۹ واحدونو له ساحې ۱۴ بشپړ واحدونه،
\LR{\latinfont OLP-0312–0324} او \LR{\latinfont OLP-0609}، په بشپړ
ډول بيا کتل شوي؛ ۲۸۵ لا پاتې دي۔ نوې کتنه د ټول کتاب د بشپړ
تصويب ادعا نۀ ده۔ سرچينه‌يي تشې او کره محدود سمونونه په اړوندو
ځايونو کښې په څرګندو پښتو يادونو کښې ښودل شوي دي۔

پاکستاني پښتو کارونه لومړيتوب لري؛ د افغانستان پښتو علمي سرچينې
د سيمه‌ييزې پرتله په څرګندو نومونو کارول شوي۔ په هغو تخصصي نومونو
کښې چې عين اصلي شاهد کم دے، پرېکړې په ښکاره د سمون وړ دي۔
د انساني ژبپوه يا رياضيپوه تصويب، د اصلي پروژې رسمي تاييد يا تړاو
نۀ ادعا کېږي۔ منجمد سرچينيزه نسخه:
\LR{\latinfont\texttt{9620cc73f9c8e0ad003c514a5d3748f29611c4c0}}۔

هم ورسره خوندي دي۔ د انځورونو بشپړې پښتو توضيحات او اصلي
