\olsection{د نورو قاعدو سموالی}

وايو چې يوه قاعده د مدلونو د يوې ټولګې لپاره سمه ده که د
!!a{tableau} هره څانګه چې د هغې ټولګې په کوم مدل کښې پوره کېږي،
د قاعدې له پلي کېدو وروسته ترې جوړه شوې څانګه هم د هماغې
ټولګې په کوم مدل کښې پوره شي۔

\begin{prop}
  \ollabel{prop:soundness-T} 
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ او }%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  \iftag{notprvBox,notprvDiamond}{د انعکاسي مدلونو لپاره سمه قاعده ده}{د انعکاسي مدلونو لپاره سمې قاعدې دي}۔
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{د T$\Box$ قاعدې په پلي کولو څانګه غځېږي؛ مقدمه
  $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$ ده:
  \iftag{probBox}{تمرین۔}{پر څانګه نوی !!{signed
      formula}~$\sFmla{\True}{!B}[\sigma]$ راځي۔ فرض کړئ
    $\mSat{M}{\Gamma}[f]$؛ په ځانګړې توګه، $\mSat{M}{\Box
      !B}[f(\sigma)]$. څرنګه چې $R$ انعکاسي ده، نو
    $Rf(\sigma)f(\sigma)$.  نو $\mSat{M}{!B}[f(\sigma)]$, يعنې
    $\mModel{M}, f$ د $\sFmla{\True}{!B}[\sigma]$ نښه لرونکے فارمول پوره کوي۔}}{}
\tagitem{prvDiamond}{د T$\Diamond$ قاعدې په پلي کولو څانګه غځېږي؛ مقدمه
  $\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$ ده:
  \iftag{probDiamond}{تمرین۔}{پر څانګه نوی !!{signed
      formula}~$\sFmla{\False}{!B}[\sigma]$ راځي۔ فرض کړئ
    $\mSat{M}{\Gamma}[f]$؛ په ځانګړې توګه، $\mSat/{M}{\Diamond
      !B}[f(\sigma)]$. څرنګه چې $R$ انعکاسي ده، نو
    $Rf(\sigma)f(\sigma)$.  نو $\mSat/{M}{!B}[f(\sigma)]$, يعنې
    $\mModel{M}, f$ د $\sFmla{\False}{!B}[\sigma]$ نښه لرونکے فارمول پوره کوي۔}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  د \olref[nml][tab][msn]{prop:soundness-T} ثبوت بشپړ کړئ۔
\end{prob}
\tagendprob

\begin{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{د 4$\Box$ قاعدې په پلي کولو څانګه غځېږي؛ مقدمه

    $\mSat{M}{\Box !B}[f(\sigma)]$. څرنګه چې $f$ په~$\mModel{M}$ کښې د څانګې د مخکښونو تفسير دے او
