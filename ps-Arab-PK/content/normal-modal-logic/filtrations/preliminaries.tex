\olsection{لومړني تعريفونه}

فلټريشنونه راته دا لار برابروي چې د مودالي منطق د خپلو نظامونو د پرېکړې
وړتيا، د هغوی د \emph{متناهي مدل خاصيت} په ښودلو سره، ثابته کړو؛ يعنې هر
هغه !!{formula} چې په يوه مدل کښې رښتينے (نارښتينے) وي، په يوه
\emph{متناهي} مدل کښې هم رښتينے (نارښتينے) وي۔ فلټريشنونه د هغو
!!{formula}s د سټونو په نسبت تعريفېږي چې د فرعي فارمولونو لپاره تړلي وي۔

\begin{defn}\ollabel{defn:modallyclosed}
  د !!{formula}s يو سټ~$\Gamma$ د \emph{فرعي فارمولونو لپاره تړلے} دے که
  په~$\Gamma$ کښې د هر !!a{formula} هر فرعي فارمول هم ولري۔ سربېره پر
  دې، $\Gamma$ \emph{مودالي تړلے} دے که د فرعي فارمولونو لپاره تړلے وي
  او $!A \in \Gamma$ دا هم لازم کړي چې $\Box!A,
  \Diamond!A \in \Gamma$۔
\end{defn}

د بېلګې په توګه، د يوه !!a{formula}~$!A$ ټول فرعي !!{formula}s
د فرعي !!{formula}s لپاره تړلے سټ جوړوي۔ کله چې د~$!A$ د فرعي
!!{formula}s د سټ له مخې د يوه مدل فلټريشن تعريفوو، نو دا به زموږ
غوښتلے خاصيت ولري: $!A$ هله او يوازې هله رښتينے (نارښتينے) کوي
چې اصلي مدل ئې هم رښتينے (نارښتينے) کوي۔

د~$\Gamma$ له مخې د~$\mModel{M}$ د فلټريشن د نړۍو سټ د لاندې
هم‌ارزښتۍ د اړيکې د ټولو ټولګيو د سټ په توګه تعريفېږي۔

\begin{defn}
فرض کړئ $\mModel{M} =\tuple{W, R, V}$ او $\Gamma$ د فرعي
!!{formula}s لپاره تړلے دے۔ پر~$W$ اړيکه~$\equiv$ داسې تعريفوو چې
دوه نړۍې هله پکښې په اړيکه وي چې د~$\Gamma$ هماغه !!{formula}s
رښتيني کړي، يعنې:
\[
u \equiv v \quad \text{هله او يوازې هله چې } \quad
\forall !A \in \Gamma : \mSat{M}{!A}[u] \Leftrightarrow \mSat{M}{!A}[v].
\]
د نړۍ~$w$ د هم‌ارزښتۍ ټولګے~$[w]_\equiv$، يا په لنډ ډول $[w]$،
د هغو ټولو نړۍو سټ دے چې له~$w$ سره د~$\equiv$ له مخې همارزې دي:
\[
[w] = \Setabs{v}{v \equiv w}.
\]
\end{defn}

\begin{prop}
  د~$\mModel{M}$ او~$\Gamma$ لپاره، پورته تعريف شوې اړيکه~$\equiv$
  د هم‌ارزښتۍ اړيکه ده، يعنې انعکاسي، متناظره او متعديه ده۔
\end{prop}

\begin{proof}
  اړيکه~$\equiv$ انعکاسي ده، ځکه $w$ د~$\Gamma$ عين هماغه
  !!{formula}s رښتيني کوي چې پخپله ئې رښتيني کوي۔ متناظره ده، ځکه
  که $u$ د~$\Gamma$ هماغه !!{formula}s رښتيني کړي چې $v$ ئې
  رښتيني کوي، نو دا خبره د~$v$ او~$u$ لپاره هم سمه ده۔ متعديه هم ده:
  که $u$ د~$\Gamma$ هماغه !!{formula}s رښتيني کړي چې $v$ ئې کوي،
  او $v$ هماغه فارمولونه رښتيني کړي چې $w$ ئې کوي، نو $u$ به
  د~$\Gamma$ هماغه !!{formula}s رښتيني کړي چې $w$ ئې کوي۔
\end{proof}

اړيکه~$\equiv$، لکه هره هم‌ارزښتۍ اړيکه، $W$ په \emph{وېش‌برخو}
وېشي، يعنې د~$W$ په هغو فرعي سټونو چې دوه دوه بېل وي او يوځای ټول~$W$
وپوښي۔ هره نړۍ $w \in W$ د يوې وېش‌برخې !!a{element} ده، يعنې
د~$[w]$، ځکه $w \equiv w$۔ نو وېش‌برخې $[w]$ ټول~$W$ پوښي۔ دا
وېش‌برخې دوه دوه بېلې دي: که $u \in [w]$ او $u \in [v]$ وي، نو
$u \equiv w$ او $u \equiv v$؛ د تقارن او تعدّي له مخې
$w \equiv v$، او ځکه $[w] = [v]$۔
