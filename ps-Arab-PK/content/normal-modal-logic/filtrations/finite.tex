\olsection{فلټريشنونه متناهي دي}

موږ د هر هغه سټ~$\Gamma$ لپاره فلټريشنونه تعريف کړي چې د فرعي
!!{formula}s لپاره تړلے وي۔ پخپله تعريف دا نۀ تضمينوي چې
فلټريشنونه متناهي دي۔ په حقيقت کښې، که $\Gamma$ نامتناهي وي (د
بېلګې په توګه، د ټولو !!{formula}s سټ وي)، فلټريشن هم نامتناهي
کېدے شي۔ خو که $\Gamma$ متناهي وي (د بېلګې په توګه، د يوه ټاکلي
!!{formula}~$!A$ د فرعي !!{formula}s سټ وي)، نو د~$\Gamma$ له
مخې هر فلټريشن هم متناهي وي۔

\begin{prop}\ollabel{prop:filt-are-finite}
  که $\Gamma$ متناهي وي، نو د مدل~$\mModel{M}$ هر
  فلټريشن~$\mModel{M^*}$ د~$\Gamma$ له مخې هم متناهي وي۔
\end{prop}

\begin{proof}
  د~$W^*$ کچه د هم‌ارزۍ اړيکې~$\equiv$ له مخې د بېلابېلو
  ټولګيو~$[w]$ شمېر دے۔ په داسې يوه ټولګي کښې هرې دوې نړۍې
  $u$ او $v$، يعنې هرې هغه $u$ او $v$ چې~$u \equiv v$ وي،
  د~$\Gamma$ د ټولو !!{formula}s~$!A$ په اړه سره يوه خوله دي:
  که $!A \in \Gamma$ وي، نو يا $!A$ په دواړو $u$ او $v$ کښې
  رښتينے وي، يا په هېڅ يوه کښې نۀ وي۔ نو هر ټولگے~$[w]$ د~$\Gamma$
  له يوه فرعي سټ سره سمون لري، يعنې د ټولو هغو $!A \in
  \Gamma$ سټ سره چې $!A$ د~$[w]$ په نړۍو کښې رښتينے وي۔
  دوه بېلابېل ټولګي $[u]$ او $[v]$ د~$\Gamma$ له يوه او هماغه
  فرعي سټ سره سمون نۀ شي لرلے۔ ځکه که په~$u$ کښې د رښتينو
  !!{formula}s سټ او په~$v$ کښې د رښتينو !!{formula}s سټ يو
  وي، نو $u$ او $v$ د~$\Gamma$ په ټولو فارمولونو کښې سره يوه
  خوله دي، يعنې $u \equiv v$۔ خو بيا $[u] = [v]$ وي۔ نو
  له~$W^*$ څخه $\Pow{\Gamma}$ ته !!a{injective} تابع شته، او
  ځکه $\card{W^*} \le \card{\Pow{\Gamma}}$۔ له دې امله که
  $\Gamma$ د $n$ جملو لرونکے وي، د~$W^*$ شمېر له~$2^n$
  څخه زيات نۀ دے۔
\end{proof}
