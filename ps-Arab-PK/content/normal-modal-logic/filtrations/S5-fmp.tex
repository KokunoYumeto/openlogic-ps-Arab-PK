\olsection{\Log{K} او \Log{S5} د متناهي مدل خاصيت لري}

\begin{defn}
  د مودالي منطق يو نظام~$\Sigma$ د \emph{متناهي
    مدل خاصيت} لري که هرکله چې کوم !!a{formula}~$!A$ د~$\Sigma$
  په يوه مدل کښې په يوه نړۍ رښتينے وي، نو $!A$ د~$\Sigma$ په
  يوه \emph{متناهي} مدل کښې هم په يوه نړۍ رښتينے وي۔
\end{defn}

\begin{prop}\ollabel{prop:K-fmp}
  \Log{K} د متناهي مدل خاصيت لري۔
\end{prop}

\begin{proof}
  \Log{K} د معتبرو !!{formula}s سټ دے، يعنې هر مدل د~\Log{K}
  مدل دے۔ د \olref[fil]{thm:filtrations} له مخې، که
  $\mSat{M}{!A}[w]$ وي، نو د~$!A$ د فرعي !!{formula}s د
  سټ~$\Gamma$ له مخې د~$\mModel{M}$ د هر فلټريشن لپاره
  $\mSat{M^*}{!A}[{[w]}]$ وي۔ په منجمده سرچينه کښې دلته د
  فلټريشن د ټولګي پر ځای اصلي نړۍ ليکل شوې وه؛ د قضيې له مخې
  هم‌ارز ټولگے اړين دے۔ هر !!{formula} يوازې متناهي ډېر فرعي
  !!{formula}s لري؛ نو $\Gamma$ متناهي دے۔ د
  \olref[fin]{prop:filt-are-finite} له مخې
  $\card{W^*} \le 2^n$، چېرته چې $n$ په~$\Gamma$ کښې د
  !!{formula}s شمېر دے۔ او ځکه \Log{K} پر مدلونو هېڅ قيد نۀ ږدي،
  $\mModel{M^*}$ د~\Log{K} مدل دے۔
\end{proof}

د فلټريشنونو له لارې د دې ښودلو لپاره چې يو منطق~\Log{L} د متناهي
مدل خاصيت لري، اړينه ده چې د~\Log{L} د يوه مدل فلټريشن پخپله هم
د~\Log{L} مدل وي۔ ډېر وخت دا ښۀ ډېر کار غواړي، او هر فلټريشن
د~\Log{L} مدل نۀ جوړوي۔ خو د عمومي مدلونو لپاره دا خاصيت پاتې
کېږي۔

\begin{prop}\ollabel{prop:univ-fin}
  فرض کړئ $\mClass{U}$ د عمومي مدلونو ټولگے وي (وګورئ
  \olref[frd][es5]{prop:S5=univ}) او $\mClass{U}_\mathrm{Fin}$ د
  ټولو متناهي عمومي مدلونو ټولگے وي۔ بيا هر !!{formula}~$!A$
  په~$\mClass{U}$ کښې هله او يوازې هله معتبر دے چې
  په~$\mClass{U}_\mathrm{Fin}$ کښې معتبر وي۔
\end{prop}

\begin{proof}
  متناهي عمومي مدلونه عمومي مدلونه دي؛ نو له کيڼ نه ښي لوري ته
  خبره څرګنده ده۔ د سرچپه لوري لپاره فرض کړئ چې~$!A$ په يوه
  عمومي مدل~$\mModel{M}$ کښې په کومه نړۍ~$w$ ناسم وي۔
  $\Gamma$ داسې واخلئ چې $!A$ او د هغه ټول فرعي فارمولونه پکښې
  وي؛ ښکاره ده چې $\Gamma$ متناهي دے۔ د~$\mModel{M}$ يو
  فلټريشن~$\mModel{M^*}$ واخلئ؛ بيا
  $\mModel{M^*}$ د \olref[fin]{prop:filt-are-finite} له مخې متناهي
  دے، او د \olref[fil]{thm:filtrations} له مخې $!A$
  په~$\mModel{M^*}$ کښې په~$[w]$ ناسم دے۔ اوس يوازې دا پاتې دي
  چې $\mModel{M^*}$ هم عمومي دے: هر $u$ او $v$ چې واخلئ، د فرض
  له مخې $Ruv$ وي او د
  \olref[fil]{defn:filtration}\olref[fil]{defn:filtration-R} له مخې
  $R^*[u][v]$ هم وي۔
\end{proof}

\begin{cor}\ollabel{cor:S5fmp}
  \Log{S5} د متناهي مدل خاصيت لري۔
\end{cor}

\begin{proof}
  د \olref[frd][es5]{prop:S5=univ} له مخې، که $!A$ په کوم
  انعکاسي او اقليدسي مدل کښې په يوه نړۍ رښتينے وي، نو په يوه
  عمومي مدل کښې هم په يوه نړۍ رښتينے وي۔ د
  \olref{prop:univ-fin} له مخې، دا په يوه متناهي عمومي مدل کښې هم
  په يوه نړۍ رښتينے وي (يعنې د~$!A$ د فرعي !!{formula}s د سټ
  له مخې د هغه مدل په فلټريشن کښې)۔ هر عمومي مدل انعکاسي او
  اقليدسي هم دے؛ نو $!A$ په يوه متناهي انعکاسي اقليدسي مدل
  کښې په يوه نړۍ رښتينے دے۔
\end{proof}

\begin{prob}
  وښيئ چې د هر مسلسل يا انعکاسي مدل هر فلټريشن هم په ترتيب سره
  مسلسل يا انعکاسي وي۔
\end{prob}

\begin{prob}
  د يوه متناظر (متعدي، اقليدسي) مدل داسې يو فلټريشن ومومئ چې
  متناظر (متعدي، اقليدسي) نۀ وي۔
\end{prob}
