\olsection{د چوکاټ تعريف‌وړتيا}

که څه هم د \olref[acc]{thm:soundschemas} سرچپه استلزامونه د مدلونو
دپاره سم نۀ دي، خو د «مدل» پر ځاے د «چوکاټ» په راوړلو سمېږي: د
\olref[acc]{thm:soundschemas} د خاصيتونو په اړه دا خبره \emph{سمه}
ده چې که !!a{formula} په يوه \emph{چوکاټ} کښې معتبر وي، نو د هغه
چوکاټ د لاسرسي اړيکه اړوند خاصيت لري۔ نو څېړل شوي !!{formula}s
د هغو چوکاټونو ټولګي \emph{تعريفوي} چې اړوند خاصيت لري۔

\begin{defn}
  که $\mClass{F}$ د چوکاټونو يو ټولګے وي، وايو چې $!A$
  دا ټولګے~$\mClass{F}$ \emph{تعريفوي}، هله او يوازې هله چې
  $\mModel{F} \Entails !A$ د ټولو او يوازې هغو
  چوکاټونو~$\mModel{F} \in \mClass{F}$ دپاره وي۔
\end{defn}

اوس د چوکاټونو د تعريف‌وړتيا بشپړې پايلې ثابتوو۔

\begin{thm}\ollabel{thm:fullCorrespondence}
که د \olref[acc]{tab:five} د ښي خوا !!{formula} په يوه
چوکاټ~$\mModel{F}$ کښې معتبر وي، نو $\mModel{F}$ د چپې خوا
خاصيت لري۔
\end{thm}

\begin{proof}
  \begin{enumerate}
  \item فرض کړئ \Ax{D} په $\mModel{F} = \tuple{W, R}$ کښې معتبره
    وي؛ يعنې $\mModel{F} \Entails \Box p \lif \Diamond p$۔
    $\mModel{M} = \tuple{W, R, V}$ پر~$\mModel{F}$ ولاړ يو مدل
    او $w \in W$ ونيسئ۔ بايد وښيو چې داسې $v$ شته چې~$Rwv$ وي۔
    فرض کړئ داسې نۀ وي: بيا د هر~$!A$ دپاره، او په ځانګړي ډول د~$p$
    دپاره، $\mSat{M}{\Box !A}[w]$ او
    $\mSat/{M}{\Diamond !A}[w]$ دواړه وي۔ خو بيا
    $\mSat/{M}{\Box p \lif \Diamond p}[w]$، چې له دې فرض سره
    په ټکر کښې دے چې $\mModel{F} \Entails \Box p \lif \Diamond p$۔
    \emph{د سرچينې يادونه: په اصلي ثبوت کښې د بکس د صدق په نښه کښې
    د نړۍ ارګومېنټ پاتې شوے و؛ دلته د همدې نړۍ ارګومېنټ ورزيات شوے دے۔}
  \item فرض کړئ \Ax{T} په $\mModel{F}$ کښې معتبره وي؛ يعنې
    $\mModel{F} \Entails \Box p \lif p$۔ $w \in W$ يوه خپلسره
    نړۍ ونيسئ؛ بايد $Rww$ وښيو۔ $u \in V(p)$ هله او يوازې هله
    وټاکئ چې $Rwu$ وي (که $q$ له~$p$ څخه بېل وي، $V(q)$ خپلسرے
    دے؛ د بېلګې په توګه $V(q) = \emptyset)$۔ فرض کړئ
    $\mModel{M} = \tuple{W, R, V}$۔ د ټاکنې له مخې، هر هغه $u$ چې
    $Rwu$ وي $\mSat{M}{p}[u]$ پوره کوي؛ نو
    $\mSat{M}{\Box p}[w]$۔ خو د فرض له مخې $\Box p \lif p$
    په~$w$ کښې رښتيا دے؛ نو $\mSat{M}{p}[w]$، او دا د~$V$ د
    تعريف له مخې يوازې هله شونې ده چې~$Rww$ وي۔
  \item د دعوې نقيض لور ثابتوو: فرض کړئ $\mModel{F}$ متناظره نۀ وي؛
    ښيو چې \Ax{B}، يعنې $p \lif \Box\Diamond p$، په
    $\mModel{F}= \tuple{W, R}$ کښې معتبره نۀ ده۔ که
    $\mModel{F}$ متناظره نۀ وي، داسې $u$، $v \in W$ شته چې
    $Ruv$ وي خو $Rvu$ نۀ وي۔ $V$ داسې تعريف کړئ چې
    $w \in V(p)$ هله او يوازې هله چې $Rvw$ نۀ وي (او $V$ په نورو
    ځايونو کښې خپلسرے دے)۔ فرض کړئ $\mModel{M} =\tuple{W, R,
      V}$۔ اوس د~$V$ له تعريفه د هر هغه $w$ دپاره چې $Rvw$ نۀ
    وي $\mSat{M}{p}[w]$، په ځانګړي ډول $\mSat{M}{p}[u]$، ځکه
    چې $Rvu$ نۀ وي۔ همدارنګه، ځکه چې $Rvw$ هله او يوازې هله چې
    $w \notin V(p)$، داسې هېڅ~$w$ نشته چې $Rvw$ او
    $\mSat{M}{p}[w]$ دواړه وي؛ نو
    $\mSat/{M}{\Diamond p}[v]$۔ ځکه چې~$Ruv$، نو
    $\mSat/{M}{\Box\Diamond p}[u]$ هم۔ له دې څخه
    $\mSat/{M}{p \lif \Box\Diamond p}[u]$ راځي؛ نو $\Ax{B}$
    په~$\mModel{F}$ کښې معتبره نۀ ده۔
  \item فرض کړئ \Ax{4} په $\mModel{F} = \tuple{W,R}$ کښې معتبره
    وي؛ يعنې $\mModel{F} \Entails \Box p \lif \Box\Box p$۔
    خپلسرې نړۍوې $u$، $v$، $w \in W$ داسې ونيسئ چې $Ruv$ او
    $Rvw$ وي؛ بايد $Ruw$ وښيو۔ $V$ داسې تعريف کړئ چې
    $z \in V(p)$ هله او يوازې هله چې $Ruz$ وي (او $V$ په نورو
    ځايونو کښې خپلسرے دے)۔ فرض کړئ $\mModel{M} =\tuple{W, R, V}$۔
    د~$V$ د تعريف له مخې، د هر هغه $z$ دپاره چې $Ruz$ وي
    $\mSat{M}{p}[z]$، نو $\mSat{M}{\Box p}[u]$۔ خو د فرض له
    مخې \Ax{4}، $\Box p \lif \Box \Box p$، په~$u$ کښې رښتيا
    ده؛ نو $\mSat{M}{\Box \Box p}[u]$۔ ځکه چې $Ruv$ او $Rvw$،
    نو $\mSat{M}{p}[w]$؛ خو د~$V$ د تعريف له مخې دا يوازې
    هله شونې ده چې $Ruw$ وي، لکه څنګه چې مو غوښتل۔
  \item بيا د دعوې نقيض لور اخلو: فرض کړئ چوکاټ
    $\mModel{F} = \tuple{W, R}$ اقليدسي نۀ وي، او ښيو چې
    \Ax{5} پکې نارښتيا کېږي؛ يعنې
    $\mModel{F} \Entails/ \Diamond p \lif \Box\Diamond p$۔
    فرض کړئ داسې نړۍوې $u$، $v$، $w \in W$ شته چې $Rwu$ او
    $Rwv$ وي، خو $Ruv$ نۀ وي۔ $V$ داسې تعريف کړئ چې د هرې نړۍ
    $z$ دپاره $z \in V(p)$ هله او يوازې هله چې $Ruz$
    \emph{نۀ} وي۔ فرض کړئ $\mModel{M} =\tuple{W, R, V}$۔ نو د
    فرض له مخې $\mSat{M}{p}[v]$، او ځکه چې $Rwv$، نو
    $\mSat{M}{\Diamond p}[w]$ هم۔ خو داسې نړۍ $y$ نشته چې
    $Ruy$ او $\mSat{M}{p}[y]$ دواړه پکې وي؛ نو
    $\mSat/{M}{\Diamond p}[u]$۔ ځکه چې $Rwu$، له دې څخه
    $\mSat/{M}{\Box\Diamond p}[w]$ راځي؛ نو \Ax{5}،
    $\Diamond p \lif \Box\Diamond p$، په~$w$ کښې نارښتيا ده۔
  \end{enumerate}
\end{proof}

د \Ax{D} د ثبوت او نورو موردونو ترمنځ توپير به مو ليدلے وي: د
ارزښت ټاکنې~$V$ هېڅ يادونه ونۀ شوه۔ په حقيقت کښې مو وښودله چې که
$\mSat{M}{\Ax{D}}$ وي، نو $\mModel{M}$ مسلسل دے۔ نو \Ax{D}
يوازې د چوکاټونو نۀ، بلکې د مسلسل \emph{مدلونو} ټولګے تعريفوي۔

\begin{cor}\ollabel{prop:D-serial}
  هر هغه مدل مسلسل دے چې \Ax{D} پکې رښتيا وي۔
\end{cor}

\begin{cor}
د \olref[acc]{tab:five} د ښي خوا هر !!{formula} د هغو چوکاټونو
ټولګے تعريفوي چې د چپې خوا خاصيت لري۔
\end{cor}

\begin{proof}
  په \olref[acc]{thm:soundschemas} کښې مو ثابته کړه چې که يو مدل
  د چپې خوا خاصيت ولري، د ښي خوا !!{formula} پکې رښتيا وي۔ نو که
  يو چوکاټ~$\mModel{F}$ د چپې خوا خاصيت ولري، د ښي خوا
  !!{formula} په~$\mModel{F}$ کښې معتبر دے۔ په
  \olref{thm:fullCorrespondence} کښې مو سرچپه استلزامونه ثابت کړل:
  که د ښي خوا !!a{formula} په~$\mModel{F}$ کښې معتبر وي، نو
  $\mModel{F}$ د چپې خوا خاصيت لري۔
\end{proof}

\begin{prob}
وښيئ چې که د \olref[nml][frd][acc]{tab:anotherfive} د ښي خوا
!!{formula} په يوه چوکاټ~$\mModel{F}$ کښې معتبر وي، نو
$\mModel{F}$ د چپې خوا خاصيت لري۔ د دې دپاره داسې چوکاټ واخلئ
چې د چپې خوا خاصيت \emph{نۀ} پوره کوي، او مناسب~$V$ داسې
وټاکئ چې د ښي خوا !!{formula} په کومه نړۍ کښې نارښتيا وي۔
\end{prob}

\olref{thm:fullCorrespondence} دا هم ښيي چې خاصيتونه يوځاے کېدلے
شي: د بېلګې په توګه، که \Ax{B} او \Ax{4} دواړه په~$\mModel{F}$
کښې معتبرې وي، نو چوکاټ هم متناظره او هم انتقالي دے، او داسې نور۔
ډېر مهم مودالي منطقونه د هغو ټولو چوکاټونو د معتبرو !!{formula}s د
مجموعې په توګه پېژندل کېږي چې څو چوکاټي خاصيتونه يوځاے لري؛ نو دوى
د هغو ټولو چوکاټونو د معتبرو !!{formula}s په توګه هم پېژندل کېدلے
شي چې اړوند تعريفوونکي !!{formula}s پکې معتبر وي۔ د بېلګې په توګه،
دوديز نظام \Log{S4} د هغو ټولو !!{formula}s مجموعه ده چې په ټولو
انعکاسي او انتقالي چوکاټونو کښې معتبر وي؛ يعنې په هغو ټولو کښې چې
\Ax{T} او~\Ax{4} دواړه معتبرې وي۔ \Log{S5} د هغو ټولو فارمولونو
مجموعه ده چې په ټولو انعکاسي، متناظرو او اقليدسي چوکاټونو کښې معتبر
وي؛ يعنې په هغو ټولو کښې چې \Ax{T}، \Ax{B} او \Ax{5} ټولې
معتبرې وي۔

د~$R$ د خاصيتونو ترمنځ منطقي اړيکې عموماً د اړوندو تعريفوونکو
!!{formula}s ترمنځ اړيکو سره سمون لري۔ د بېلګې په توګه، هره انعکاسي
اړيکه مسلسل هم ده؛ نو چې کله \Ax{T} په يوه چوکاټ کښې معتبره وي،
\Ax{D} هم پکې معتبره وي۔ (پام وکړئ چې دا اړيکه د استلزام اړيکه
\emph{نۀ} ده۔ دا خبره سمه نۀ ده چې هر کله
$\mSat{M}{\Ax{T}}[w]$ وي، نو $\mSat{M}{\Ax{D}}[w]$ هم وي۔)
څو داسې اړيکې دلته ثبتوو۔

\begin{prop}\ollabel{prop:relation-facts}
  فرض کړئ $R$ پر يوه سټ $W$ دوه‌ځاييزه اړيکه ده؛ بيا:
  \begin{enumerate}
  \item که $R$ انعکاسي وي، نو مسلسل هم ده۔
  \item که $R$ متناظره وي، نو هله او يوازې هله انتقالي ده چې
    اقليدسي وي۔
  \item که $R$ متناظره يا اقليدسي وي، نو کمزورې هم‌لوري ده
    («الماسي خاصيت» لري)۔
  \item که $R$ اقليدسي وي، نو کمزورې تړلې ده۔
  \item که $R$ تابعي وي، نو مسلسل ده۔  
  \end{enumerate}
\end{prop}

\begin{prob}
  \olref[nml][frd][def]{prop:relation-facts} ثابته کړئ۔
\end{prob}
