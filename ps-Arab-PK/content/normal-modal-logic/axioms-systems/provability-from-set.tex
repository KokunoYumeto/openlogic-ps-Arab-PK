\olsection{د \usetoken{P}{formula} له يوه سټ څخه \usetoken{S}{derivability}}

په \olref[prf][prs]{sec} کښې مو په يوه نظام~$\Sigma$ کښې د
!!a{formula} د ثابتېدو مفهوم تعريف کړے و۔ اوس دا مفهوم په~$\Sigma$
کښې د !!{formula}s له سټ~$\Gamma$ څخه ثابتېدو ته غځوو۔

\begin{defn}\ollabel{defn:Gammaproves}
  !!^a{formula}~$!A$ په نظام~$\Sigma$ کښې د !!{formula}s له سټ~$\Gamma$ څخه
  (چې په $\Gamma \Proves[\Sigma] !A$ ليکل کېږي) هله او يوازې هله
  !!{derivable} وي، چې داسې $!B_1$, \dots,
  $!B_n \in \Gamma$ وي چې
  $\Sigma \Proves !B_1 \lif (!B_2 \lif \cdots (!B_n \lif !A)
  \cdots)$.
\end{defn}
