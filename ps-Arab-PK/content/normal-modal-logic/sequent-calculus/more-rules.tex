\begin{ex}

مثلاً \iftag{prvBox}{$\Diamond\Box !A \lif !A$}{$!A \lif
\Box\Diamond !A$}، چې په~$\Log{S5}$ کښې معتبر دے، بې له~\Cut
څخه ثبوت نۀ لري۔ دلته د~\Cut په کارولو يو !!a{derivation} دے:
\begin{prooftree}
\iftag{prvBox}{\iftag{prvDiamond}{% both [] and <> primitive

\RightLabel{T$\Box$}
\UnaryInf$\Box!A \fCenter !A$

\RightLabel{\RightR{\lif}}
\UnaryInf$\fCenter \Diamond\Box!A \lif !A$

\Axiom$\Box !A \fCenter \Box!A$

\RightLabel{\RightR{\Exchange}}
\UnaryInf$\fCenter \Box\lnot\Box!A, \Box!A$

\Axiom$!A \fCenter !A$
\RightLabel{T$\Box$}

\RightLabel{\Cut}

\begin{prob}
د لاندې ادعاوو د ښودلو لپاره سېکوېنټ !!{derivation}s وړاندې کړئ:
  \begin{enumerate}
  \item $\Log{KT5} \Proves \Ax{B}$;
  \item $\Log{KT5} \Proves \Ax{4}$;
  \item $\Log{KDB4} \Proves \Ax{T}$;
  \item $\Log{KB4} \Proves \Ax{5}$;
  \item $\Log{KB5} \Proves \Ax{4}$;
  \item $\Log{KT} \Proves \Ax{D}$.
  \end{enumerate}
\end{prob}
