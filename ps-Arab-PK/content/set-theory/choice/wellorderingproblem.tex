\olsection{د ښه ترتيب مسئله}

ښکاره ده چې د ښه ترتيب منل يا نه منل ډېرې پايلې لري۔
خو د دې اصل بحث تر ډېره د هغه پر يوه همارز اصل، يعنې
د ټاکنې پر اصل، راڅرخي۔ نو اوس همدغه اصل څېړو
(او همارزي به يې ثابته کړو)۔

په \citeyear{Cantor1883} کښې کانتور د ښه ترتيب
له اصله ملاتړ وکړ۔ هغه ښه ترتيب د استدلال له بنسټونو څخه ګاڼه، ځکه د اصل کارونه عامه وه او ګڼې پايلې ترې راوتلې۔ دا دريځ \citeauthor{Potter2004} په خپل کتاب کښې رااخلي
\citeyear[p.~243]{Potter2004}۔
خو کانتور بالاخره دې باور ته ورسېد چې دا «اصل»
ثبوت غواړي۔ د رياضي پوهانو ټولنه هم ورته نظر ته ورسېده۔

دا مسئله په \citeyear{Zermelo1904} کښې زېرملو
«هواره» کړه۔ د هغه د حل د بيانولو دپاره يو څو
تعريفونو ته اړتيا لرو۔

\begin{defn}
تابعه $f$ د \emph{انتخاب تابعه} هله او يوازې هله ده
چې $f(x) \in x$ د هر $x \in \dom{f}$ دپاره وي۔
وايو چې $f$ \emph{د $A$ دپاره د ټاکنې تابعه} ده
هله او يوازې هله چې $f$ د ټاکنې تابعه وي او
$\dom{f} = A \setminus \{\emptyset\}$ وي۔
\end{defn}

په وجداني توګه، د هر (ناخالي) سټ
$x \in A$ دپاره د $A$ د ټاکنې تابعه
يو ځانګړے غړے، $f(x)$، له $x$ څخه
\emph{ټاکي}۔ نو د ټاکنې اصل دا دے:

\begin{axiom}[ټاکنه]
هر سټ د ټاکنې تابعه لري۔
\end{axiom}

زېرملو وښودله چې ټاکنه ښه ترتيب لازموي، او
ښه ترتيب هم ټاکنه لازموي:

\begin{thm}[په $\ZF$ کښې]\ollabel{thmwochoice}
ښه ترتيب او ټاکنه همارز دي۔
\end{thm}

\begin{proof}
\emph{له چپ څخه ښي ته۔} $A$ د سټونو يو سټ
ونيسئ۔ نو $\bigcup A$ د اتحاد د اصل له مخې
شتون لري، او د ښه ترتيب له مخې داسې
$<$ شته چې $\bigcup A$ ښه ترتيبوي۔ اوس
$f(x) = \text{د $<$ له مخې تر ټولو کوچنے غړے د }x$
د هر ناتش غړي دپاره وټاکئ۔ دا تابع د بېلونې له لارې
د اړوند کارتيزي حاصل دننه جوړېږي، او د $A$ دپاره
د ټاکنې تابعه ده۔

\emph{له ښي څخه چپ ته۔} $A$ وټاکئ۔ د ټاکنې له مخې
د ټاکنې تابعه $f$ د $\Pow{A} \setminus \{\emptyset\}$
دپاره شته۔ د \olref[hartogs]{HartogsLemma} له مخې د $A$
دپاره يو داسې ترتيبي عدد $h$ واخلو چې $A$ ته ترې يو پر
يو تابع نۀ وي۔ د نامتناهيت هاخوا بازګښت پر $h$ باندې
تابع په دې ډول تعريفوي:
\begin{align*}
	g(\alpha) &=
		\begin{cases}
			A &\text{که } A \subseteq \funimage{g}{\alpha}\\
			f(A \setminus \funimage{g}{\alpha}) & \text{په بل صورت کښې.}
		\end{cases}
\end{align*}
د رتبې له مخې $A\notin A$، نو خپله $A$ د تمېدو نښه ده۔
بازګښت هر نوی ارزښت د هماغه پړاو د مخکنيو ارزښتونو له
حاصله ټاکي۔ \olref[sth][spine][recursion]{transrecursionfun}
په دې ټاکلې ساحه کښې د تابعې شتون او يوازينيوالے راکوي۔

تر تمېدو مخکې، د هر $\alpha$ دپاره
$g(\alpha) =
f(A \setminus \funimage{g}{\alpha}) \in A \setminus
\funimage{g}{\alpha}$؛ نو
$g(\alpha) \notin \funimage{g}{\alpha}$۔
که $g(\alpha) = g(\beta)$ وي، نو
$g(\beta) \notin \funimage{g}{\alpha}$، يعنې
$\beta \notin \alpha$؛ او په همدې ډول
$\alpha \notin \beta$۔ د درې‌ګونيتوب له مخې
$\alpha = \beta$۔ نو د تمېدو نه مخکې محدوده شوې تابعه
!!{injective} ده۔

که تر $h$ مخکې هېڅ پړاو کښې تم نۀ شو، نو $g$ به
له $h$ نه $A$ ته !!{injective} وي، چې له
\olref[hartogs]{HartogsLemma} سره تناقض لري۔ نو داسې
لږ ترتيبي عدد $\alpha$ شته چې $A = g[\alpha]$؛
تر دغه پړاو مخکې ټول ارزښتونه د $A$ غړي دي او په
همدغه پړاو کښې يې حاصل ټول $A$ دے۔ که $A$ تش وي،
$\alpha=0$ دے۔

د $g$ لومړۍ $\alpha$ برخې ته محدودول له $\alpha$
څخه $A$ ته !!a{bijection} ورکوي۔ د ترتيبي عدد ښه
ترتيب د همدې محدودې تابعې په وسيله $A$ ته لېږدوو۔
\end{proof}

نو ښه ترتيب او ټاکنه دواړه يو ځاے ټينګېږي
يا يو ځاے لوېږي۔ خو پوښتنه لا پاتې ده: ايا
دواړه ټينګېږي که لوېږي؟
