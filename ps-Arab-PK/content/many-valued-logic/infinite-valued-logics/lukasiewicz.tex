\olsection{د لوکاشېوېچ منطق}

\begin{editorial}
  دا برخه د لوکاشېوېچ د ناطقو ارزښتونو ماتريس، صدق تابعې او له متناهي ماتريسونو سره ئې اړيکه بيانوي۔
\end{editorial}

\begin{defn}\ollabel{def:lukasiewicz} نامتناهي ارزښته د لوکاشېوېچ
منطق~$\LogLuk[\infty]$ په دې ماتريس تعريفېږي:
\begin{enumerate}
  \item معياري بياني ژبه~$\Lang L_0$ چې
  $\lnot$, $\land$, $\lor$, $\lif$ لري۔
  \item د رښتياوالي د قيمتونو سټ~$V_\infty$۔
  \item يوازې~$1$ ټاکل شوے قيمت دے؛ يعنې~$V^+ = \{1\}$۔
  \item صدق تابعې په لاندې تابعو ټاکل کېږي:
  \begin{align*}
    \tf{\lnot}[\LogLuk](x) & = 1 - x\\
    \tf{\land}[\LogLuk](x,y) & = \min(x,y)\\
    \tf{\lor}[\LogLuk](x,y) & = \max(x,y)\\
    \tf{\lif}[\LogLuk](x,y) & = \min(1,1-(x-y)) = \begin{cases}
      1 & \text{که } x \le y\\
      1-(x-y) & \text{په بل حالت کښې.}
    \end{cases}
    \end{align*}
\end{enumerate}
د لوکاشېوېچ $m$-ارزښته منطق همداسې تعريفېږي، خو~$V = V_m$ لري۔
\end{defn}

\begin{prop}
  د~\olref[thr][luk]{def:lukasiewicz} له مخې تعريف شوے منطق~$\LogLuk[3]$
  د~\olref{def:lukasiewicz} له مخې له تعريف شوي~$\LogLuk[3]$ سره يو شان دے۔
\end{prop}

\begin{proof}
  دا د~\olref[thr][luk]{def:lukasiewicz} د نښلوونکو صدق جدولونه
  د~\olref{def:lukasiewicz} له معادلو څخه له ترلاسه شوو جدولونو سره
  په پرتله کولو ليدل کېدے شي:
  \begin{center}
    \begin{tabular}{c|c} 
      $\tf{\lnot}$ & \\ 
      \hline  
      $1$ & $0$ \\ 
      $1/2$ & $1/2$ \\
      $0$ & $1$ 
    \end{tabular}
    \quad
    \begin{tabular}{c|ccc} 
      $\tf{\land}[\LogLuk[3]]$ & $1$ & $1/2$ & $0$ \\ 
      \hline 
      $1$ & $1$ & $1/2$ & $0$ \\ 
      $1/2$ & $1/2$ & $1/2$ & $0$\\ 
      $0$ & $0$ & $0$ & $0$ 
    \end{tabular}
    \\[2ex]
    \begin{tabular}{c|ccc} 
      $\tf{\lor}[\LogLuk[3]]$ & $1$ & $1/2$ & $0$ \\ 
      \hline 
      $1$ & $1$ & $1$ & $1$ \\ 
      $1/2$ & $1$ & $1/2$ & $1/2$ \\
      $0$ & $1$ & $1/2$ & $0$ 
    \end{tabular}
    \quad
    \begin{tabular}{c|ccc} 
      $\tf{\lif}[\LogLuk[3]]$ & $1$ & $1/2$ & $0$ \\ 
      \hline 
      $1$ & $1$ & $1/2$ & $0$ \\ 
      $1/2$ & $1$ & $1$ & $1/2$  \\ 
      $0$ & $1$ & $1$ & $1$ 
    \end{tabular}
  \end{center} 
\end{proof}

\begin{prop}\ollabel{prop:luk-infty-m}
  که~$\Gamma \Entails[\LogLuk[\infty]] !B$ وي، نو د هر~$m \ge 2$
  دپاره~$\Gamma \Entails[\LogLuk[m]] !B$ هم وي۔
\end{prop}

\begin{proof}
  تمرين۔
\end{proof}

\begin{prob}
  \olref[mvl][inf][luk]{prop:luk-infty-m} ثابت کړئ۔
\end{prob}

که په~$\Gamma$ او~$!B$ کښې يوازې متناهي ډېر بياني متغيرونه
راشي، نو برعکس لورے هم صدق کوي۔ که يوه ناطقه ارزښت ټاکنه ټولې
مقدمې پوره کړي او پايله پوره نۀ کړي، د هغو متغيرونو د قيمتونو
يو ګډ مخرج واخلو۔ ټول دا قيمتونه په يوه متناهي سټ کښې راځي، او
هلته هم صدق تابعې هماغه دي؛ نو هماغه مخالف مثال په متناهي
ماتريس کښې هم شته۔
د نامتناهي مقدماتو لپاره عمومي برعکس لورے ناسم دے۔ د
$!A \oplus !B$ لنډيز د~$\lnot !A \lif !B$ لپاره وکاروئ او
د~$S_n(p)$ په ليکدود د~$p$ د~$n$ نقلونو دغه مجموعه وښيئ؛
$S_1(p)=p$۔ که د~$p$ قيمت~$x$ وي، د دې مجموعې قيمت
$\min(1,nx)$ دے۔ د هر مثبت صحيح~$n$ لپاره دا مقدمه واخلئ:
\[
(p_n \lif \lnot S_n(p_n)) \land (\lnot S_n(p_n) \lif p_n).
\]
دا مقدمه هله رښتيا ده چې د دواړو خواوو قيمتونه برابر وي، يعنې
$\pAssign v(p_n)=1/(n+1)$۔ په ناطق ماتريس کښې ټولې مقدمې
پوره کېدے شي، او جلا متغير~$q$ ته صفر ورکولے شو۔ خو په
$V_m$ کښې د~$n=m-1$ مقدمه د~$1/m$ قيمت غواړي، چې په دغه
سټ کښې نشته۔ نو هر متناهي ماتريس له دغو مقدماتو څخه~$q$
لازموي، خو نامتناهي ناطق ماتريس ئې نۀ لازموي۔

نامتناهي ارزښته د لوکاشېوېچ منطق يو پېژندل شوے فازي منطق دے۔
د فازي منطق په ليکنو کښې شرطي نښلوونکے ډېر ځله په~$\lnot !A \lor !B$
تعريفېږي۔ په دې سره به يو نامتناهي ارزښته قوي کليني منطق ترلاسه شي۔

\begin{prob}
  وښيئ چې~$(p \lif q) \lor (q \lif p)$ د~$\LogLuk[\infty]$
  تاوتولوژي ده۔
\end{prob}
