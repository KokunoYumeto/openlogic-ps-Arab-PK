\olsection{څو ارزښته منطقونه د~$\LogCL$ د فرعي منطقونو په توګه}

معمول څو ارزښته منطقونه ټول د داسې ماتريسونو په وسيله
تعريفېږي چې د~$\{\True, \False\}$ د مدخلونو دپاره ئې د صدق
تابعې قيمت د کلاسيکو صدق تابعو له قيمت سره برابر وي۔ په کره توګه،
په دغو منطقونو کښې که~$x \in \{\True, \False\}$ وي، نو
$\tf{\lnot}[\Log L](x) = \tf{\lnot}[\LogCL](x)$؛ او که~$\star$
له~$\land$، $\lor$ او~$\lif$ څخه هر يو وي او
$x, y \in \{\True, \False\}$ وي، نو
$\tf{\star}[\Log L](x,y) = \tf{\star}[\LogCL](x,y)$۔ په بل عبارت،
د~$\lnot$، $\land$، $\lor$ او~$\lif$ صدق تابعې چې پر
$\{\True,\False\}$ محدودې شي، کټ مټ کلاسيکې صدق تابعې دي۔

\begin{prop}\ollabel{prop:mvl-cl}
  فرض کړئ چې د څو ارزښته منطق~$\Log L$ په ژبه کښې
  $\lnot$، $\land$، $\lor$ او~$\lif$ نښلوونکي شته،
  $\True, \False \in V$ دي او د دې منطق صدق تابعې
  لاندې شرطونه پوره کوي:
  \begin{enumerate}
    \item\ollabel{prop:not} $\tf{\lnot}[\Log L](x) = \tf{\lnot}[\LogCL](x)$ که~$x =
    \True$ يا~$x = \False$ وي؛
    \item\ollabel{prop:land} $\tf{\land}[\Log L](x,y) = \tf{\land}[\LogCL](x,y)$،
    \item\ollabel{prop:lor} $\tf{\lor}[\Log L](x,y) = \tf{\lor}[\LogCL](x,y)$،
    \item\ollabel{prop:lif} $\tf{\lif}[\Log L](x,y) = \tf{\lif}[\LogCL](x,y)$،
    که~$x, y \in \{\True, \False\}$ وي۔
  \end{enumerate}
  نو د دواړو منطقونو د ګډې ژبې د هغو فارمولونو دپاره چې هر
  کارېدونکے ثابت او نښلوونکے ئې پر دغو دوو قيمتونو يو شان صدق
  تابعه لري، هره داسې ارزښت ټاکنه~$\pAssign v$ چې~$V$ ته ځي او د هر
  بياني متغير دپاره~$\pAssign v(p) \in \{\True,\False\}$ وي،
  $\pValue v[\Log L](!A) = \pValue v[\LogCL](!A)$ پوره کوي۔
  د سرچينې د حد يادونه: د دې استقرايي پايلې دپاره د هر کارېدونکي
  ثابت او نښلوونکي صدق تابعې هم په دغو دوو قيمتونو کښې سره برابرې
  پکار دي؛ پورته شرطونه اضافي نښلوونکي يا ثابتونه نۀ رانغاړي۔
\end{prop}

\begin{proof}
  پر~$!A$ د استقرا له مخې۔
  \begin{enumerate}
  \item که~$!A \ident p$ اتمي وي، نو
  $\pValue v[\Log L](!A) = \pAssign v(p) = \pValue
  v[\LogCL](!A)$ لرو۔
  \item که~$!A \ident \lnot B$ وي، نو
  \begin{align*}
    \pValue v[\Log L](!A) & = \tf{\lnot}[\Log L](\pValue v[\Log L](!B)) & &\text{د \olref[val]{defn:pValue} له مخې}\\
    & = \tf{\lnot}[\Log L](\pValue v[\LogCL](!B)) && \text{د استقرايي فرض له مخې}\\
    & = \tf{\lnot}[\LogCL](\pValue v[\LogCL](!B))
&&\text{د \olref{prop:not} د فرض له مخې،}\\
&&&\text{ځکه چې $\pValue v[\LogCL](!B) \in \{\True, \False\}$ دے،}\\
& = \pValue v[\LogCL](!A) &&\text{د \olref[val]{defn:pValue} له مخې}.
\end{align*}
\item که~$!A \ident (!B \land !C)$ وي، نو
\begin{align*}
  \pValue v[\Log L](!A) & = \tf{\land}[\Log L](\pValue v[\Log L](!B), \pValue v[\Log L](!C)) & &\text{د \olref[val]{defn:pValue} له مخې}\\
  & = \tf{\land}[\Log L](\pValue v[\LogCL](!B),\pValue v[\LogCL](!C)) && \text{د استقرايي فرض له مخې}\\
  & = \tf{\land}[\LogCL](\pValue v[\LogCL](!B),\pValue v[\LogCL](!C))
&&\text{د \olref{prop:land} د فرض له مخې،}\\
&&&\text{ځکه چې $\pValue v[\LogCL](!B),\pValue v[\LogCL](!C) \in \{\True, \False\}$ دي،}\\
& = \pValue v[\LogCL](!A) &&\text{د \olref[val]{defn:pValue} له مخې}.
\end{align*}
\end{enumerate}
هغه حالتونه چې~$!A \ident (!B \lor !C)$ او
$!A \ident (!B \lif !C)$ وي، په همدې ډول دي۔
\end{proof}

\begin{cor}
  که يو څو ارزښته منطق د~\olref{prop:mvl-cl} شرطونه پوره کوي،
  $\True \in V^+$ او~$\False \notin V^+$ وي، نو
  د پورته بيان شوې ګډې ژبې او برابرو صدق تابعو په اړه
  ${\Entails[\Log L]} \subseteq {\Entails[\LogCL]}$؛ يعنې که
  $\Gamma \Entails[\Log L] !B$ وي، نو
  $\Gamma \Entails[\LogCL] !B$ هم وي۔ په ځانګړې توګه،
  دلته ټولې مقدمې او پايله هم د همدغو شرطونو تابع دي۔
  په همدې ګډه ژبه کښې د~$\Log L$ هره تاوتولوژي يوه
  کلاسيکه تاوتولوژي هم ده۔
\end{cor}

\begin{proof}
  نقيض عکس ثابتوو۔ فرض کړئ~$\Gamma \Entails/[\LogCL] !B$۔
  بيا يوه داسې !!{valuation}~$\pAssign v\colon \PVar \to
  \{\True, \False\}$ شته چې د هر~$!A \in \Gamma$ دپاره
  $\pValue v[\LogCL](!A) = \True$ او
  $\pValue v[\LogCL](!B) = \False$ وي۔ ځکه چې
  $\True, \False \in V$ دي، دا !!{valuation}~$\pAssign v$
  د~$\Log L$ دپاره هم !!a{valuation} ده۔ د~\olref{prop:mvl-cl}
  له مخې، د هر~$!A \in \Gamma$ دپاره
  $\pValue v[\Log L](!A) = \True$ او
  $\pValue v[\Log L](!B) = \False$ دي۔ ځکه چې
  $\True \in V^+$ او~$\False \notin V^+$ دي، له دې څخه
  $\pAssign v \Entails[\Log L] \Gamma$ او
  $\pAssign v \Entails/[\Log L] !B$ راوځي؛ يعنې
  $\Gamma \Entails/[\Log L] !B$۔
\end{proof}
