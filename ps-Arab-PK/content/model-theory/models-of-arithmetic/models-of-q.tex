\section{د~$\Th{Q}$ مدلونه}

\begin{explain}
موږ پوهېږو چې داسې نامعياري !!{structure}s شته چې هماغه
!!{sentence}s رښتونې کوي چې~$\Struct{N}$ ئې رښتونې کوي؛ يعنې د
$\Th{TA}$ مدلونه دي۔ څرنګه چې $\Sat{N}{\Th{Q}}$، د~$\Th{TA}$ هر
مدل د~$\Th{Q}$ هم مدل دے۔ $\Th{Q}$ له~$\Th{TA}$ څخه ډېره کمزورې
ده؛ مثلاً
$\Th{Q} \Proves/ \lforall[x][\lforall[y][\eq[(x + y)][(y+x)]]]$۔
کمزورې تيورۍ په اسانه پوره کېږي: ډېر مدلونه لري۔ مثلاً $\Th{Q}$
داسې مدلونه لري چې
$\lforall[x][\lforall[y][\eq[(x + y)][(y+x)]]]$ دروغوي، خو هغه د
$\Th{TA}$ يا حتا د~$\Th{PA}$ مدلونه نۀ شي کېدے۔ د~$\Th{Q}$ مدلونه
نسبتاً ساده هم دي: په څرګند ډول ئې ټاکلے شو۔
\end{explain}

\begin{ex}
\ollabel{ex:model-K-of-Q}
هغه !!{structure}~$\Struct{K}$ ونيسئ چې تعريف ساحه ئې
$\Domain{K} = \Nat \cup \{a\}$ او تفسيرونه ئې دا وي:
\begin{align*}
  \Assign{\Obj{0}}{K} & = 0\\
  \Assign{\prime}{K}(x) & =
  \begin{cases}
    x+1 & \text{که $x\in \Nat$ وي}\\
    a & \text{که $x = a$ وي}
  \end{cases}\\
  \Assign{+}{K}(x, y) & =
  \begin{cases}
    x+y & \text{که $x$، $y \in\Nat$ وي}\\
    a & \text{په نورو حالاتو کښې}
  \end{cases}\\
  \Assign{\times}{K}(x, y) & =
  \begin{cases}
    xy & \text{که $x$، $y \in\Nat$ وي}\\
    0 & \text{که $x = 0$ يا $y = 0$ وي}\\
    a & \text{په نورو حالاتو کښې}\\
  \end{cases}\\
  \Assign{<}{K} & =
  \Setabs{\tuple{x,y}}{x, y \in \Nat \text{ او } x<y} \cup
  \Setabs{\tuple{x,a}}{x \in \Domain{K}}
\end{align*}
د~$\Sat{K}{\Th{Q}}$ ښودلو دپاره بايد ثابته کړو چې د~$\Th{Q}$ ټول
بديهي اصول په~$\Struct{K}$ کښې رښتوني دي۔ د اسانتيا دپاره
$x^\nssucc$ د~$\Assign{\prime}{K}(x)$ په ځاے ليکو (په~$\Struct{K}$
کښې د~$x$ «جانشين»)، $x \nsplus y$ د~$\Assign{+}{K}(x,y)$ په ځاے
(په~$\Struct{K}$ کښې د~$x$ او~$y$ «مجموعه»)، $x \nstimes y$ د
$\Assign{\times}{K}(x,y)$ په ځاے (په~$\Struct{K}$ کښې د~$x$
او~$y$ «حاصل ضرب»)، او
$x \nsless y$ د~$\tuple{x,y}\in\Assign{<}{K}$ په ځاے ليکو۔ په دې
لنډونو سره د~$\Struct{K}$ عمليات په لا څرګند ډول داسې ښودلے شو:
\[
\begin{array}{c|c}
  x & x^\nssucc \\
  \hline
  n & n+1 \\
  a & a
\end{array}
\qquad
\begin{array}{c|ccc}
  x \nsplus y & 0 & m & a \\
  \hline
  0 & 0 & m & a \\
  n & n & n+m & a \\
  a & a & a & a \\
\end{array}
\qquad
\begin{array}{c|ccc}
  x \nstimes y & 0 & m & a \\
  \hline
  0 & 0 & 0 & 0 \\
  n & 0 & nm & a \\
  a & 0 & a & a \\
\end{array}
\]
$n\nsless m$ هله او يوازې هله چې $n<m$، د~$n$، $m\in\Nat$
دپاره؛ او د هر~$x\in\Domain{K}$ دپاره $x\nsless a$۔

$\Sat{K}{\lforall[x][\lforall[y][(\eq[x'][y'] \lif \eq[x][y])]]}$،
ځکه $\nssucc$ !!{injective} ده۔
$\Sat{K}{\lforall[x][\eq/[\Obj 0][x']]}$، ځکه په~$\Struct{K}$ کښې
$0$ د~$\nssucc$ جانشين نۀ دے۔
$\Sat{K}{\lforall[x][(\eq[x][\Obj 0] \lor
\lexists[y][\eq[x][y']])]}$، ځکه د هر~$n>0$ دپاره
$n=(n-1)^\nssucc$ او $a=a^\nssucc$۔

$\Sat{K}{\lforall[x][\eq[(x + \Obj 0)][x]]}$، ځکه
$n\nsplus0=n+0=n$ او د~$\nsplus$ له تعريفه $a\nsplus0=a$۔
$\Sat{K}{\lforall[x][\lforall[y][\eq[(x + y')][(x + y)']]]}$ لږه
پېچلې ده۔ که $n$ او~$m$ دواړه معياري وي:
\begin{align*}
(n \nsplus m^\nssucc) & = (n+(m+1)) = (n+m)+1 = (n \nsplus m)^\nssucc
\intertext{ځکه $\nsplus$ او $^\nssucc$ پر معياري شمېرو له~$+$ او
  $\prime$ سره سلا دي۔ اوس فرض کړئ $x \in \Domain{K}$۔ بيا}
(x \nsplus a^\nssucc) & = (x \nsplus a) = a = a^\nssucc = (x \nsplus a)^\nssucc
\intertext{پاتې حالت دا دے چې $y \in \Domain{K}$ وي خو $x=a$۔
  دلته هم بايد وڅېړو چې $y=n$ معياري دے يا $y=a$:}
(a \nsplus n^\nssucc) & = (a \nsplus (n+1)) = a = a^\nssucc = (a \nsplus n)^\nssucc\\
(a \nsplus a^\nssucc) & = (a \nsplus a) = a = a^\nssucc = (a \nsplus a)^\nssucc
\end{align*}
\textbf{د ژباړې د نسخې سمون (OLMOD-008):} سرچينې په دې وېش کښې
د y وروستے حالت b ليکلے و، خو د K د تعريف ساحه يوازې طبيعي شمېرې
او a لري؛ دلته له ورپسې معادلې سره سم وروستے حالت a ليکل شوے دے۔
البته دا له اړتيا زيات تفصيل دے۔ مثلاً څرنګه چې د هر~$z$ دپاره
$a\nsplus z=a$، نو سمدستي $a\nsplus a^\nssucc=a$ پايله کولے شو۔
پاتې بديهي اصول هم په همدې ډول کتل کېدے شي۔

نو $\Struct{K}$ د~$\Th{Q}$ يو مدل دے۔ «جمع»~$\nsplus$ ئې تبادلي هم
ده۔ خو داسې نورې جملې شته چې په~$\Struct{N}$ کښې رښتونې او په
$\Struct{K}$ کښې دروغ دي، او برعکس۔ مثلاً $a\nsless a$، نو
$\Sat{K}{\lexists[x][x < x]}$ او
$\Sat/{K}{\lforall[x][\lnot x<x]}$۔ له دې څرګندېږي چې
$\Th{Q} \Proves/ \lforall[x][\lnot x < x]$۔
\end{ex}

\begin{prob}
ثابته کړئ چې د~\olref[mod][mar][mdq]{ex:model-K-of-Q} جوړښت
$\Struct{K}$ د~$\Th{Q}$ پاتې بديهي اصول پوره کوي:
\begin{align*}
  & \lforall[x][\eq[(x \times \Obj 0)][\Obj 0]] \tag{$!Q_6$}\\
  & \lforall[x][\lforall[y][\eq[(x \times y')][((x \times y) + x)]]] \tag{$!Q_7$}\\
  & \lforall[x][\lforall[y][(x < y \liff \lexists[z][\eq[(z' + x)][y]])]] \tag{$!Q_8$}
\end{align*}
داسې !!a{sentence} ومومئ چې يوازې~$\prime$ ولري، په~$\Struct{N}$
کښې رښتونې او په~$\Struct{K}$ کښې دروغ وي۔
\end{prob}

\begin{ex}
\ollabel{ex:model-L-of-Q} هغه !!{structure}~$\Struct{L}$ ونيسئ چې
تعريف ساحه ئې $\Domain{L} = \Nat \cup \{a,b\}$ او تفسيرونه
$\Assign{\prime}{L} = \nssucc$، $\Assign{+}{L} = \nsplus$ د لاندې
جدولونو له مخې وي:
\[
\begin{array}{c|c}
  x & x^\nssucc \\
  \hline
  n & n+1 \\
  a & a\\
  b & b
\end{array}
\qquad
\begin{array}{c|ccc}
  x \nsplus y & m & a & b\\
  \hline
  n & n+m & b & a\\
  a & a & b & a\\
  b & b & b & a
\end{array}
\]
څرنګه چې $\nssucc$ !!{injective} ده، $0$ ئې د قيمتونو په ساحه کښې
نۀ دے، او له~$0$ پرته هر $x\in\Domain{L}$ پکې دے، نو
$!Q_1$--$!Q_3$ په~$\Struct{L}$ کښې رښتوني دي۔ د هر~$x$ دپاره
$x\nsplus0=x$، نو $!Q_4$ هم رښتونے دے۔ د~$!Q_5$ دپاره
$x\nsplus y^\nssucc$ او $(x\nsplus y)^\nssucc$ پرتله کړئ۔ که $x$
او~$y$ دواړه معياري وي، برابر دي، ځکه بيا $\nssucc$ او~$\nsplus$
له~$\prime$ او~$+$ سره سلا دي۔ که $x$ نامعياري او~$y$ معياري وي،
$x\nsplus y^\nssucc=x=x^\nssucc=(x\nsplus y)^\nssucc$۔ که $x$ او
$y$ دواړه نامعياري وي، څلور حالتونه لرو:
\begin{align*}
&  a \nsplus a^\nssucc  = b = b^\nssucc = (a \nsplus a)^\nssucc\\
&  b \nsplus b^\nssucc  = a = a^\nssucc = (b \nsplus b)^\nssucc\\
&  b \nsplus a^\nssucc  = b = b^\nssucc = (b \nsplus a)^\nssucc\\
&  a \nsplus b^\nssucc  = a = a^\nssucc = (a \nsplus b)^\nssucc\\
\intertext{که $x$ معياري خو $y$ نامعياري وي، لرو:}
&  n \nsplus a^\nssucc  = n \nsplus a = b = b^\nssucc = (n \nsplus a)^\nssucc\\
&  n \nsplus b^\nssucc  = n \nsplus b = a = a^\nssucc = (n \nsplus b)^\nssucc
\end{align*}
\textbf{د ژباړې د نسخې سمون (OLMOD-009):} سرچينې د درېيم نامعياري
حالت په ښي اړخ کښې ناتړلے y ليکلے و؛ دلته د همدې کرښې له چپ اړخ
او د جدول له مخې a بېرته ايښوول شوے دے۔
نو $\Sat{L}{!Q_5}$۔ خو $a\nsplus0\neq0\nsplus a$، نو
$\Sat/{L}{\lforall[x][\lforall[y][\eq[(x+y)][(y+x)]]]}$۔
\end{ex}

\begin{prob}
د~\olref[mod][mar][mdq]{ex:model-L-of-Q} جوړښت~$\Struct{L}$ په
$\nstimes$ او~$\nsless$ سره وغځوئ، څو د~$\times$ او~$<$ تفسيرونه
وي۔ وښيئ چې جوړښت مو د~$\Th{Q}$ پاتې بديهي اصول پوره کوي:
\begin{align*}
& \lforall[x][\eq[(x \times \Obj 0)][\Obj 0]] \tag{$!Q_6$}\\ &
  \lforall[x][\lforall[y][\eq[(x \times y')][((x \times y) + x)]]]
  \tag{$!Q_7$}\\ & \lforall[x][\lforall[y][(x < y \liff \lexists[z][\eq[(z'+x)][y]])]] \tag{$!Q_8$}
\end{align*}
\end{prob}

\begin{prob}
د~\olref[mod][mar][mdq]{ex:model-L-of-Q} په~$\Struct{L}$ کښې
$a^\nssucc=a$ او $b^\nssucc=b$ دي۔ ايا د~$\Th{Q}$ داسې مدل شته چې
$a^\nssucc=b$ او $b^\nssucc=a$ وي؟
\end{prob}

\begin{explain}
موږ د~$\Th{Q}$ داسې مدلونه په څرګند ډول جوړ کړل چې نامعياري
!!{element}s ئې د معياري عناصرو «هاخوا» پراتۀ دي۔ په حقيقت کښې،
بديهي اصول همدومره غواړي۔ يو نامعياري !!{element}~$x$ د~${} \nsless 0$ شرط
پوره کولے نۀ شي، ځکه $\Th{Q} \Proves
\lforall[x][\lnot x<0]$ (وګورئ
\olref[inc][req][min]{lem:less-zero})۔ همدارنګه د هر~$n$ دپاره
$\Th{Q} \Proves \lforall[x][(x < \num{n}' \lif
(\eq[x][\num{0}] \lor \eq[x][\num{1}] \lor \dots \lor
\eq[x][\num{n}]))]$ (\olref[inc][req][min]{lem:less-nsucc})، نو د
هېڅ~$n>0$ دپاره $a\nsless n$ نۀ شو لرلے۔
\end{explain}
