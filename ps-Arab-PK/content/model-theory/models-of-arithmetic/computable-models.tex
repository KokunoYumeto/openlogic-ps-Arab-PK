\section{د حساب محاسبه کېدونکي مدلونه}

\begin{explain}
معياري مدل~$\Struct{N}$ دوه ښې ځانګړتياوې لري۔ تعريف ساحه ئې طبيعي
شمېرې~$\Nat$ دي، يعنې عناصر ئې هماغه ډول څيزونه دي چې د حساب
په ژبه ئې په هکله خبرې کول غواړو، او معياري عددنښه~$\num{n}$ په
رښتيا~$n$ په نښه کوي۔ بله ښه ځانګړتيا دا ده چې د~$\Lang{L_A}$ د
غيرمنطقي نښو ټول تفسيرونه \emph{محاسبه کېدونکي} دي۔ د جانشين، جمع
او ضرب تابعې چې $\Assign{\prime}{N}$، $\Assign{+}{N}$ او
$\Assign{\times}{N}$ دي، د شمېرو محاسبه کېدونکې تابعې دي (مثلاً د
ټيورينګ ماشينونو له خوا محاسبه کېدونکې، يا په ابتدايي بازګښت
تعريفېدونکې)۔ او په~$\Struct{N}$ کښې د کوچنيوالي اړيکه، يعنې
$\Assign{<}{N}$، پرېکړه کېدونکې ده۔

د~$\Th{Q}$ او~$\Th{PA}$ په شان د حسابي تيوريو نامعياري مدلونه بايد
نامعياري عناصر ولري۔ نو تعريف ساحې ئې عموماً له~$\Nat$ سره نور
!!{element}s هم لري۔ خو هر شمېرېدونکے !!{structure} پر هر
!!{denumerable} سټ، د~$\Nat$ په ګډون، جوړېدے شي۔ نو داسې نامعياري
مدلونه هم شته چې تعريف ساحه ئې~$\Nat$ وي۔ البته په داسې
مدل~$\Struct{M}$ کښې لږ تر لږه ځينې شمېرې خپل معمولي نقش نۀ شي
لوبولے، ځکه داسې~$k$ بايد وي چې د ټولو~$n \in \Nat$ دپاره
له~$\Value{\num{n}}{M}$ څخه بېل وي۔
\end{explain}

\begin{defn}
د~$\Lang{L_A}$ يو !!^a{structure}~$\Struct{M}$ \emph{محاسبه
کېدونکے} دے هله او يوازې هله چې $\Domain{M}=\Nat$؛
$\Assign{\prime}{M}$، $\Assign{+}{M}$ او~$\Assign{\times}{M}$
محاسبه کېدونکې تابعې؛ او $\Assign{<}{M}$ پرېکړه کېدونکې اړيکه وي۔
\end{defn}

\begin{ex}\ollabel{ex:comp-model-q}
د~\olref[mdq]{ex:model-K-of-Q} جوړښت~$\Struct{K}$ په ياد راوړئ۔
تعريف ساحه ئې $\Domain{K}=\Nat\cup\{a\}$ او تفسيرونه ئې دا وو:
\begin{align*}
  \Assign{\Obj{0}}{K} & = 0\\
  \Assign{\prime}{K}(x) & =
  \begin{cases}
    x+1 & \text{که $x\in \Nat$ وي}\\
    a & \text{که $x = a$ وي}
  \end{cases}\\
  \Assign{+}{K}(x, y) & =
  \begin{cases}
    x+y & \text{که $x$، $y \in\Nat$ وي}\\
    a & \text{په نورو حالاتو کښې}
  \end{cases}\\
  \Assign{\times}{K}(x, y) & =
  \begin{cases}
    xy & \text{که $x$، $y \in\Nat$ وي}\\
    0 & \text{که $x=0$ يا $y=0$ وي}\\
    a & \text{په نورو حالاتو کښې}\\
  \end{cases}\\
  \Assign{<}{K} & =
  \Setabs{\tuple{x,y}}{x, y \in \Nat \text{ او } x<y} \cup
  \Setabs{\tuple{x,a}}{x \in \Domain{K}}
\end{align*}
\textbf{د ژباړې د نسخې سمون (OLMOD-014):} سرچينې د وروستي سټ په
شرط کښې n ليکلے و، حال دا چې په مرتبه جوړه کښې x راغلے دے؛ دلته
شرط د x د تعريف ساحې غړيتوب ته سم شوے، لکه د K په اصلي تعريف کښې
چې راغلے دے۔
خو $\Domain{K}$ !!{denumerable} او له~$\Nat$ سره همشمېره دے۔
مثلاً $g\colon\Nat\to\Domain{K}$ چې $g(0)=a$ او د~$n>0$ دپاره
$g(n)=n-1$ وي، يوه !!a{bijection} ده۔ دا د~$\Th{Q}$ د نوي
مدل~$\Struct{K'}$ او~$\Struct{K}$ تر منځ پر آيزومورفيزم بدلولے شو۔
په~$\Struct{K'}$ کښې بايد د~$\Lang{L_A}$ نښو ته بېلې تابعې او
اړيکې وټاکو، ځکه د~$\Nat$ بېل !!{element}s د معياري او نامعياري
شمېرو نقشونه لوبوي۔
\textbf{د ژباړې د نسخې سمون (OLMOD-015):} سرچينې د مثبت n دپاره
د n جمع يو نقشه ليکلې وه، چې صفر او يو د تابعې له قيمتونو بهر
پرېږدي او دوه اړخيزه يو په يو نۀ ده۔ دلته د n منفي يو نقشه ليکل
شوې؛ همدا نقشه د ورپسې ټولو لېږدول شويو تفسيرونو سره سلا ده۔

په ځانګړي ډول، اوس~$0$ د~$a$ نقش لوبوي، نه د تر ټولو کوچنۍ معياري
شمېرې۔ تر ټولو کوچنۍ معياري شمېره اوس~$1$ ده۔ نو
$\Assign{\Obj{0}}{K'}=1$ ټاکو۔ د جانشين تابع هم اوس بېله ده: يوه
معياري شمېره، يعنې $n>0$، ورکړل شي، بيا هم~$n+1$ راګرځوي۔ خو~$0$
اوس د~$a$ نقش لوبوي، چې خپل جانشين دے۔ نو
$\Assign{\prime}{K'}(0)=0$۔ د جمع او ضرب دپاره هم لرو:
\begin{align*}
\Assign{+}{K'}(x, y) & =
  \begin{cases}
    x+y-1 & \text{که $x$، $y >0$ وي}\\
    0 & \text{په نورو حالاتو کښې}
  \end{cases}\\
  \Assign{\times}{K'}(x, y) & =
  \begin{cases}
    1 & \text{که $x = 1$ يا $y = 1$ وي}\\
    xy - x - y + 2 & \text{که $x$، $y > 1$ وي}\\
    0 & \text{په نورو حالاتو کښې}\\
  \end{cases}
\end{align*}
او $\tuple{x,y}\in\Assign{<}{K'}$ هله او يوازې هله چې $x<y$،
$x>0$ او $y>0$ وي، يا $y=0$ وي۔

دا ټولې تابعې د طبيعي شمېرو محاسبه کېدونکې تابعې دي او
$\Assign{<}{K'}$ پر~$\Nat$ يوه پرېکړه کېدونکې اړيکه ده؛ خو دغه
تابعې پر~$\Nat$ د جانشين، جمع او ضرب له معمولي تابعو سره يو شان نۀ
دي، او $\Assign{<}{K'}$ هم پر~$\Nat$ له معمولي~$<$ سره يو شان نۀ ده۔
\end{ex}

\begin{prob}
داسې !!a{structure}~$\Struct{L'}$ ورکړئ چې $\Domain{L'}=\Nat$ وي او
د~\olref[mod][mar][mdq]{ex:model-L-of-Q} له~$\Struct{L}$ سره
آيزومورف وي۔
\end{prob}

\begin{explain}
\Olref{ex:comp-model-q} ښيي چې $\Th{Q}$ پر~$\Nat$ د تعريف ساحې
محاسبه کېدونکي نامعياري مدلونه لري۔ خو لاندې پايله ښيي چې د
$\Th{PA}$ د مدلونو دپاره دا خبره رښتونې نۀ ده (او ځکه د~$\Th{TA}$
د مدلونو دپاره هم نۀ ده)۔
\end{explain}

\begin{thm}[د ټېننباوم قضيه]
$\Struct{N}$ تر آيزومورفيزم پورې د~$\Th{PA}$ يوازينے محاسبه
کېدونکے مدل دے۔
\textbf{د ژباړې د نسخې سمون (OLMOD-016):} سرچينې «يوازينے» په
لفظي ډول ويلي وو، خو د طبيعي شمېرو پر بنسټ محاسبه کېدونکي او
آيزومورف بيا-نومول شوي نقلونه شته؛ د ټېننباوم د قضيې دقيقه پايله دا ده چې
هېڅ محاسبه کېدونکے \emph{نامعياري} مدل نشته۔ دلته «تر
آيزومورفيزم پورې» دا دقيقوالی څرګندوي۔
\end{thm}
