\olsection{پرېکړې وړتيا}
پام وکړئ چې د بشپړتيا د قضيې ثبوت د هر
$\Gamma \Proves/ !A$ دپاره د نامتناهي شمېر نړۍو لرونکے داسې مدل
راکوي چې $\Gamma \Entails/ !A$ پرې ښيي۔ لاندې قضيه وايي چې
د $\Entails !A$ د ثابتولو دپاره د ټولو متناهي مدلونو (يعنې د
متناهي نړۍو سټ لرونکو مدلونو) لپاره د $\mSat{M}{!A}$ ثابتول
کافي دي۔

\begin{thm}\ollabel{thm:decidability}
  که $\Entails/ !A$ وي، نو داسې متناهي مدل شته چې
  $\mSat/{M'}{!A}$ وي۔
\end{thm}

\begin{proof}
  فرض کړئ $\mModel{M}=\tuple{W, R, V}$ داسې وي چې
  $\mSat/{M}{!A}$۔ تعريف شوي نښلوونکي پر خپل تعريف پرانيزو او
  د~$!A$ د ټولو فرعي فارمولونو سټ~$S$ اخلو؛ په دې سټ کښې
  فارمول پخپله او ټول راغلي !!{propositional variable}s هم دي۔
  د هرې نړۍ د رښتينو فرعي فارمولونو سټ داسې ليکو:
  \[
    t(w) = \Setabs{!B \in S}{\mSat{M}{!B}[w]}.
  \]
  $\mModel{M'}=\tuple{W', R', V'}$ داسې تعريفوو چې
  $W' = \Setabs{t(w)}{w \in W}$ وي، $R'tu$ هله او يوازې
  هله وي چې $t \subseteq u$ وي، او
  $V'(p)=\Setabs{t \in W'}{p \in t}$ وي۔ $W'$ ناتش دے او
  تر~$2^{|S|}$ زيات غړي نۀ لري۔ د سټونو شمول جزئي ترتيب دے
  او په دې ترتيب د اټومي متغير غړيتوب پاتې کېږي۔ نو
  $\mModel{M'}$ !!a{relational model} دے۔

  پر فرعي !!{formula}s~$!B \in S$ استقرا کوو او په هرې
  نړۍ کښې دا برابري ثابتوو:
  \[
    \mSat{M}{!B}[w] \text{ هله او يوازې هله چې } \mSat{M'}{!B}[t(w)].
  \]
  د !!{propositional variable}s حالت د ټاکنې له تعريفه راځي،
  او محال په دواړو مدلونو کښې په هېڅ نړۍ کښې صدق نۀ لري۔
  د عطف او فصل دواړه غړي هم په~$S$ کښې دي، نو د هغو د صدق
  په اړه د استقرا فرض د عطف او فصل صدق هم ساتي۔ اوس
  $!B = !C \lif !D$ وګورئ۔ که $\mSat{M}{!B}[w]$ وي او
  $t(w)\subseteq t(v)$ وي، نو شرط په~$t(v)$ کښې هم غړے
  دے او په اصلي نړۍ~$v$ کښې رښتينے دے۔ که مخکښ فارمول
  په نوي مدل کښې په~$t(v)$ کښې رښتينے وي، د استقرا له مخې
  په اصلي نړۍ~$v$ کښې هم رښتينے دے۔ د انعکاسيت له مخې
  شرط په هماغې نړۍ کښې پلي کوو او د استقرا په کارولو
  يې پايله بېرته~$t(v)$ ته وړو۔ نو شرط په نوې نړۍ~$t(w)$
  کښې رښتينے دے۔ که $\mSat/{M}{!B}[w]$ وي، داسې $v$ شته
  چې $Rwv$، $\mSat{M}{!C}[v]$ او $\mSat/{M}{!D}[v]$ وي۔
  د صدق د پاتې کېدو له مخې $t(w)\subseteq t(v)$ دے او
  د استقرا له مخې په~$t(v)$ کښې مخکښ رښتينے او پايله
  نارښتينې ده۔ نو شرط په~$t(w)$ کښې هم نارښتينے دے۔
  اوس $!A \in S$ دے۔ په اصلي مدل کښې هره نړۍ چې $!A$
  پکښې نارښتينے وي، په متناهي مدل کښې يې انځور هم
  $!A$ نارښتينے کوي۔
\end{proof}

\begin{prob}
د \olref[int][sc][dec]{thm:decidability} په استقرا کښې
د محال، اټومي متغير، عطف او فصل حالتونه په تفصيل وليکئ۔ د
دغو !!{formula}s~$!B \in S$ دپاره وښيئ چې
$\mSat{M}{!B}[w]$ هله او يوازې هله وي چې
$\mSat{M'}{!B}[t(w)]$ وي۔ بيا داسې دوې نړۍ ومومئ چې
يو شان اټومي ارزښتونه لري، خو يو شرط په يوه نړۍ کښې
رښتينے او په بله کښې نارښتينے وي۔
\end{prob}

له \olref{thm:decidability} څخه دا راځي چې د~$\Entails !A$
د پرېکړې دپاره الګوريتم شته۔ تر پورته متناهي حده ټول
جزئي ترتيبونه او د اړوندو متغيرونو ټولې يورخيزې ټاکنې
شمېرئ، او په هر مدل کښې د فارمول صدق وګورئ۔
