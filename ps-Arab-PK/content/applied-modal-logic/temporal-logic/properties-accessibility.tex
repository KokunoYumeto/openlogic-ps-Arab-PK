\olsection{د زماني چوکاټونو خاصيتونه}

زموږ زماني مدلونه پر اړيکه~$\prec$ هېڅ شرط نۀ ږدي۔ د يوه
لوري لپاره د عادي مودالي اکسيومونو له ډلې~$K$ په ټولو مدلونو
کښې سم دے؛ د راتلونکي او تېر لوري بڼې ئې $K_{\Gtemp}$ او
~$K_{\Htemp}$ دي۔ د دواړو لورو د يوې اړيکې په توګه کارول نورې
ګډې معتبرې طرحې هم ورکوي: د هرې راتلونکې شېبې په تېر کښې
اوسنۍ شېبه شته، او د هرې تېرې شېبې په راتلونکي کښې اوس شته۔
\begin{align*}
\tag{$K_{\Gtemp}$}  \Gtemp (p \to q) & \to (\Gtemp p \to \Gtemp q)\\
\tag{$K_{\Htemp}$}  \Htemp (p \to q) & \to (\Htemp p \to \Htemp q)
\end{align*}

خو که غواړو زموږ مدلونه د وخت د ښودلو په طريقه سخت شرطونه ولري،
لکه دا چې~$\prec$ يو خطي ترتيب وي، نو په مدلونو کښې به نورې
معتبرې طرحې هم ولرو۔

\begin{table}[t]
  \begin{tabular}{| p{.48\textwidth} || p{.48\textwidth} |}
    \hline
    {\emph{که $\prec$ \dots وي}} & {\emph{نو \dots په~$\mModel{M}$ کښې رښتيا دے:}} \\
    \hline \hline
    \emph{انتقالي}: \newline
    $\forall u \forall v \forall w ((u \prec v \land v \prec w) \lif u \prec w)$ &
    $\Ftemp \Ftemp p \lif \Ftemp p$  \\
    \hline
    \emph{د هرې جوړې پرتله کېدنه}: \newline
    $\forall w \forall v (w \prec v \lor w = v \lor v \prec w)$ &
    $(\Ftemp \Ptemp  p \lor \Ptemp \Ftemp p) \lif (\Ptemp p \lor p \lor \Ftemp p)$\\
    \hline
    \emph{ګڼه}: \newline
    $\forall w \forall v (w \prec v \to \exists u(w \prec u \land u \prec v))$ &
    $\Ftemp p \lif \Ftemp \Ftemp p$ \\
    \hline
    \emph{په تېر لوري بې سرحده}: \newline
    $\forall w \exists v( v \prec w)$ &
    $\Htemp p \to \Ptemp p$ \\
    \hline
    \emph{په راتلونکي لوري بې سرحده}: \newline
    $\forall w \exists v( w \prec v)$ &
    $\Gtemp p \to \Ftemp p$ \\
    \hline
  \end{tabular}
  \caption{د زماني چوکاټونو د مطابقت ځينې خاصيتونه۔}
  \ollabel{tab:correspondence}
\end{table}

د \olref{tab:correspondence} ځينې خاصيتونه ښايي د وخت د ښودونکي
مدل لپاره مطلوب ښکاره شي۔ د جدول طرحه دا غواړي چې هره جوړه چې
يوه ګډه مخکنۍ يا وروستۍ شېبه لري، سره د پرتله کېدو وړ وي۔
مثلاً په دوو جلا شېبو او تشې اړيکې کښې طرحه معتبره ده،
او دواړه شېبې سره د پرتله کېدو اړيکه نۀ لري۔ خو که څۀ هم پر~$\prec$ اړيکه هر ډول
شرط ايښودلے شو، ټول شرطونه د زماني منطق په ژبه کښې د څرګندېدونکو
!!{formula}s سره نۀ برابرېږي۔ د بېلګې په توګه، نفي انعکاسيت،
يعنې دا چې $\forall w \lnot (w \prec w)$، په همدې بنسټيزه
يو‌ځاييزه ژبه کښې د چوکاټ د اعتبار له لارې نۀ تعريفېږي۔
په يوه ځان ته رسېدونکې شېبه کښې د متغيرونو هره ارزښت‌ټاکنه
د صحيح عددونو پر هر ټکي يو شان وټاکئ او هلته کلک ترتيب واخلئ۔
د تېر او راتلونکي لوري په دواړو چوکاټونو کښې ناتش دي۔ نو په
استقرا سره د هر فارمول رښتياوالے په دواړو کښې يو شان وي۔
هره طرحه چې پر کلک ترتيب معتبره وي، پر ځان ته رسېدونکې
يوې شېبې هم معتبره ده۔
