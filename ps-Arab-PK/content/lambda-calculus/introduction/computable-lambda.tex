\olsection{محاسبه کېدونکې تابعې \usetoken{S}{lambda definable} دي}

\begin{thm}
\ollabel{thm:computable-lambda}
هره محاسبه کېدونکې جزوي تابع !!{lambda definable} ده۔
\end{thm}

\begin{proof}
غواړو وښيو چې جزوي محاسبه کېدونکې تابع يو لامبډا ترم !!{lambda defined} کوي۔ د \olref[cmp][rec][nft]{thm:kleene-nf} له مخې کافي ده چې د يوې ټولې بنسټيزې بازګشتي ازموينې لومړنے بريالے شاهد ولټوو او د شاهد له کوډ څخه ځواب واخلو۔ د څو ورودي عددونو په حالت کښې لومړے د هغوی کوډ جوړوو؛ د لړۍ کوډول بنسټيز بازګښت دے۔ د لومړنيو تابعو، د ټوليزو تابعو د ترکيب او د ټوليزو تابعو د بنسټيز بازګښت لاندې جوړونې ښيي چې لومړنۍ تابعې !!{lambda definable} دي او په دغو دوو عملونو هره ټوله !!{lambda definable} تابع بيا !!{lambda definable} پاتې کېږي۔ د ازموينې ځانګړونکې تابعه داسې اخلو چې د بريا پر وخت صفر او کنه يو ورکوي۔ د شاهد څخه د ځواب اخستونکے ترم~$U$ دې وي، او ازموينه دې~$P$ !!{lambda defined} کړي۔ د ورودي عددونو لړۍ په~$\vec x$ ښيو۔
د چرچ د عددنښو لپاره دا ترمونه وکاروئ:
\[
\mathsf{T}=\lambda a.\lambda b.a,\quad\mathsf{F}=\lambda a.\lambda b.b,\quad Z=\lambda n.n(\lambda u.\mathsf{F})\mathsf{T}.
\]
پر صفر~$Z$ لومړے انتخاب او پر هر مثبت عدد دويم انتخاب راګرځوي۔ $S$ د تالي ترم دے او~$Y$ د ثابت ټکي د تورينګ ترم دے۔ د ورکړو ورودي عددونو لپاره دا لټون جوړوو:
\[
L_{\vec x}=Y(\lambda h.\lambda s.(Z(P(\vec x,s)))(U(s))(h(Ss))).
\]
د تعريفوونکي ترم بدنه~$L_{\vec x}\num{0}$ ده؛ د ورودي لړۍ متغيرونه په بهرني تجريد کښې تړو۔ د هرې شېبې په پيل کښې د ثابت ټکي د لګېدلو تر ټولو کيڼ رېډکس راکموو۔ بيا ازموينه د صفر يا يو عددنښې ته رسوو او اړوند انتخاب اخلو۔ د بريا په صورت کښې~$U$ د شاهد ځواب ورکوي۔ د ناکامۍ په صورت کښې د تالي عددنښه اخلو او د هماغه لټون بل پړاو ته ځو۔ دا د لومړني بريالي شاهد دپاره د غوښتل شوې پايلې عددنښه ورکوي، نو په دې حالت کښې تابع همدا ترم !!{lambda defined} کوي۔
که شاهد هېڅکله پيدا نۀ شي، همدا راکمونه نامتناهي ده۔ د هغې په هر پړاو کښې لږ تر لږه يو تر ټولو کيڼ رېډکس راکمېږي۔ د \href{https://pages.jh.edu/rrynasi1/FoundationsOFMath/Literature/LambdaCalculus/Barendregt1984TheLambdaCalculus.ItsSyntax+Semantics.RevisedEdition.pdf}{بارېنډرېخټ، ۱۹۸۴، قضيه ۸٫۴٫۹ او پايله ۸٫۴٫۱۰} له مخې، داسې نامتناهي لړۍ لرونکے ترم عادي بڼه نۀ لري۔ نو په ناتعريفه ننوت کښې هم د لامبډا-تعريف شرط پوره کېږي۔
\end{proof}

د ثبوت پاتې برخه د لا آسانه لوستلو دپاره به د نښو يو لا دوديز
طرز وکاروو۔ مثلاً، د $Mxyz$ پر ځاے $M(x, y, z)$ ليکو۔ که څۀ هم دا
نښه څو آرګومېنټونه راښيي، په ياد ولرئ چې د بې‌ټايپه لامبډا حساب
ترمونه ځانته ټاکلې ځاييزه کچه نۀ لري۔ نو د هماغه ترم~$M$ دپاره د
$M(x,y)$ او $M(x,y,z,w)$ ليکل هم معنا لري۔ خو دا نښه ښيي چې موږ
$M$ د درې متغيرونو د تابعې په توګه ګورو، او د تعريفونو تر شا موخه
روښانوي۔ همداسې به د ``$M$ په $M =
\lambd[x][\lambd[y][\lambd[z][\dots]]]$ تعريف کړئ'' پر ځاے
وايو ``$M$ په $M(x,y,z) = \dots$ تعريف کړئ''۔
