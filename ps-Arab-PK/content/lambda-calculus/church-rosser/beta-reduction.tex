\olsection{$\beta$-راکمول}

\begin{lem}\ollabel{lem:one-par}
  که $M \bredone M'$ وي، نو $M \bredpar M'$.
\end{lem}
\begin{proof} که $M \bredone M'$ وي او انقباض د ترم په سر کښې
    وي، نو $M$ د ځينو $x$، $N$ او~$Q$ دپاره
    $(\lambd[x][N])Q$ دے او $M'$، $\Subst{N}{Q}{x}$ دے۔
    د \olref[pb]{thm:refl} له مخې $N \bredpar N$ او $Q \bredpar Q$
    دي؛ نو له \olref[pb]{defn:bredpar} او
    \olref[pb]{defn:bredpar4} څخه سملاسي
    $(\lambd[x][N])Q \bredpar \Subst{N}{Q}{x}$ راځي۔
    \textbf{د سرچينې سپيناوی (OLLAM-039):} اصلي ثبوت يوازې
    د سر رېډکس حالت اخلي؛ د يوګامي انقباض سازګاره تړلتيا د
    تجريد او تطبيق دننه حالتونه هم لري، چې د ترم پر جوړښت
    په استقرا او د هممهاله راکمول په اړوندو قواعدو بشپړېږي۔
\end{proof}

\begin{lem}\ollabel{lem:par-red}
  که $M \bredpar M'$ وي، نو $M \bred M'$.
\end{lem}

\begin{proof} د $M \bredpar M'$ د !!{derivation} پر جوړښت استقرا کوو۔
  \begin{enumerate}
  \item وروستۍ قاعده \olref[pb]{defn:bredpar1} ده: په دې حالت کښې
    $M$ او $M'$ دواړه هماغه $x$ دي، او $x \bred x$.
  \item وروستۍ قاعده \olref[pb]{defn:bredpar2} ده: $M$ د ځينو
    $x$، $N$ او $N'$ دپاره $\lambd[x][N]$ او $M'$،
    $\lambd[x][N']$ دے، چې $N \bredpar N'$ وي۔ د استقرا له
    فرضيې $N \bred N'$ لرو۔ نو $\lambd[x][N] \bred
    \lambd[x][N']$؛ دا د $N \bred N'$ د هماغو $\bredone$
    انقباضونو لړۍ د تجريد دننه تطبيقول دي۔
  \item وروستۍ قاعده \olref[pb]{defn:bredpar3} ده: $M$،
    $PQ$ او $M'$ د ځينو $P$، $Q$، $P'$ او $Q'$ دپاره $P'Q'$
    دے، چې $P \bredpar P'$ او $Q \bredpar Q'$ وي۔ د استقرا
    له فرضيې $P \bred P'$ او $Q \bred Q'$ لرو۔ نو لومړے
    د $P \bred P'$ لړۍ او بيا د $Q \bred Q'$ لړۍ په کارولو
    $PQ \bred P'Q'$ ترلاسه کېږي۔
  \item وروستۍ قاعده \olref[pb]{defn:bredpar4} ده: $M$،
    $(\lambd[x][N])Q$ او $M'$، $\Subst{N'}{Q'}{x}$ دے؛ دا
    د ځينو $x$، $N$، $N'$، $Q$ او $Q'$ دپاره دي، چې
    $N \bredpar N'$ او $Q \bredpar Q'$ وي۔ د استقرا له
    فرضيې $Q \bred Q'$ او $N \bred N'$ ترلاسه کوو۔ نو
    $(\lambd[x][N])Q \bred \Subst{N'}{Q'}{x}$: لومړے
    $N \bred N'$، بيا $Q \bred Q'$، او په پاے کښې د
    $(\lambd[x][N'])Q'$ انقباض $\Subst{N'}{Q'}{x}$ ته۔
    \textbf{د سرچينې سمون (OLLAM-046):} د متغيرونو په
    لړليک کښې د سرچينې دويم اېم-پرایم د مخکنيو فرضيو او
    وروسته کارېدونکي اېن-پرایم پر ځاے چاپ شوے و؛ دلته
    اېن-پرایم راغلے دے۔
  \end{enumerate}
\end{proof}

\begin{lem}\ollabel{lem:str}
  $\bred$ تر ټولو کوچنۍ انتقالي اړيکه ده چې $\bredpar$ پکې
  شامله ده۔
\end{lem}

\begin{proof}
  $\xred$ دې تر ټولو کوچنۍ انتقالي اړيکه وي چې
  $\bredpar$ پکې شامله ده۔

  $\bred \subseteq \xred$: فرض کړئ $M \bred M'$، يعنې
  $M \ident M_1 \bredone \dots \bredone M_k \ident M'$۔
  د \olref{lem:one-par} له مخې $M
  \ident M_1 \bredpar \dots \bredpar M_k \ident M'$۔
  څرنګه چې $\xred$، $\bredpar$ رانغاړي او انتقالي ده،
  $M \xred M'$ ترلاسه کېږي۔ \textbf{د سرچينې سپيناوی
  (OLLAM-040):} که دا لړۍ صفر ګامونه ولري، هممهاله راکمول
  د مخکېنۍ انعکاس قضیې له مخې انعکاسي دے؛ نو د شاملېدو
  دليل په هماغه حالت کښې هم کار کوي۔

  $\xred \subseteq \bred$: فرض کړئ $M \xred M'$، يعنې $M
  \ident M_1 \bredpar \dots \bredpar M_k \ident M'$۔ د \olref{lem:par-red}
  له مخې $M \ident M_1 \bred \dots \bred M_k \ident M'$۔
  څرنګه چې $\bred$ انتقالي ده، $M \bred M'$.
\end{proof}

\begin{thm}\ollabel{thm:cr}
  $\bred$ د چرچ--روسر خاصيت لري۔
\end{thm}

\begin{proof}
  دا د \olref[dap]{thm:str}، \olref[pb]{thm:cr} او
  \olref{lem:str} سملاسي پايله ده۔
\end{proof}
