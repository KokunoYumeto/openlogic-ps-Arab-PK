\olsection{له~$\Th{Q}$ سره سازګارې تيورۍ ناپرېکړه‌وړې دي}

لاندې قضيه وايي چې يوازې~$\Th{Q}$ ناپرېکړه‌وړې نۀ ده، بلکې په
حقيقت کښې هره هغه تيوري ناپرېکړه‌وړې ده چې له~$\Th{Q}$ سره اختلاف
ونۀ لري۔

\begin{thm}
پرېږدئ~$\Th{T}$ د حساب په ژبه کښې هره هغه تيوري وي چې له~$\Th{Q}$
سره سازګاره وي (يعنې~$\Th{T} \cup \Th{Q}$ سازګاره وي)۔ بيا
$\Th{T}$ ناپرېکړه‌وړې ده۔
\end{thm}

\begin{proof}
په ياد ولرئ چې~$\Th{Q}$ د بديهي اصولو متناهي سټ~$!Q_1$، \dots،
$!Q_8$ لري۔ ان دا ټول په يوه بديهي اصل بدلولے شو:
$!E = !Q_1 \land \dots \land !Q_8$۔

فرض کړئ~$\Th{T}$ يوه د پرېکړې وړ تيوري ده چې له~$\Th{Q}$ سره
سازګاره ده۔ پرېږدئ
\[
C = \Setabs{!A}{\Th{T} \Proves !E \lif !A}.
\]
موږ ښيو چې~$C$ به د~$\Th{Q}$ او~$\Th{\bar Q}$ يوه محاسبه کېدونکې
بېلوونکې وي، چې تناقض دے۔ لومړے، که~$!A$ په~$\Th{Q}$ کښې وي، نو
$!A$ د~$\Th{Q}$ له بديهي اصولو څخه ثابتېدونکې ده؛ د استنتاج د
قضيې له مخې په لومړۍ درجې منطق کښې د~$!E \lif !A$ يو
!!a{derivation} شته۔ نو~$!A$ په~$C$ کښې ده۔

له بلې خوا، که~$!A$ په~$\Th{\bar Q}$ کښې وي، نو په لومړۍ درجې
منطق کښې د~$!E \lif \lnot !A$ يو ثبوت شته۔ که~$\Th{T}$،
$!E \lif !A$ هم ثابت کړي، نو~$\Th{T}$، $\lnot !E$ ثابتوي، او په
دې حالت کښې~$\Th{T} \cup \Th{Q}$ ناسازګاره ده۔ خو موږ فرض کوو
چې~$\Th{T} \cup \Th{Q}$ سازګاره ده، نو~$\Th{T}$، $!E \lif !A$
نۀ ثابتوي، او له دې امله~$!A$ په~$C$ کښې نۀ ده۔

موږ وښودله چې که~$!A$ په~$\Th{Q}$ کښې وي، نو په~$C$ کښې ده، او
که~$!A$ په~$\Th{\bar Q}$ کښې وي، نو په~$\Complement{C}$ کښې ده۔
نو~$C$ يوه محاسبه کېدونکې بېلوونکې ده، او همدا هغه تناقض دے چې
غوښت مو۔
\end{proof}

دا قضيه ډېره پياوړې ده۔ د بېلګې په توګه، له دې څخه لاندې پايله
لازمېږي:

\begin{cor}
  د حساب د ژبې دپاره لومړۍ درجې منطق (يعنې سټ
  $\Setabs{!A}{\text{$!A$ په لومړۍ درجې منطق کښې ثابتېدونکې ده}}$)
  ناپرېکړه‌وړ دے۔
\end{cor}

\begin{proof}
لومړۍ درجې منطق د~$\emptyset$ د پايلو سټ دے، چې له~$\Th{Q}$ سره
سازګار دے۔
\end{proof}
