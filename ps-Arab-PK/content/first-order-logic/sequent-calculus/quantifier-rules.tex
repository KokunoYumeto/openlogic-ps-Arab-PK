\olsection{د سورونو قاعدې}

\subsection{د $\lforall$ دپاره قاعدې}

\begin{defish}
\Axiom$ !A(t), \Gamma \fCenter \Delta$
\RightLabel{\LeftR{\lforall}}
\UnaryInf$ \lforall[x][!A(x)],\Gamma \fCenter \Delta$
\DisplayProof
\hfill
\Axiom$ \Gamma \fCenter \Delta, !A(a) $
\RightLabel{\RightR{\lforall}}
\UnaryInf$ \Gamma \fCenter \Delta, \lforall[x][!A(x)]$
\DisplayProof
\end{defish}

په \LeftR{\lforall} کښې $t$ يوه تړلې اصطلاح ده (يعنې داسې اصطلاح
چې متغيرونه نۀ لري)۔ په \RightR{\lforall} کښې $a$~!!a{constant} دے
چې بايد د \RightR{\lforall} قاعدې په لاندې سېکوېنټ کښې هېڅ ځاے نۀ
وي راغلے۔ موږ $a$ د \RightR{\forall} استنتاج \emph{ځانګړے متغير}
بولو۔\footnote{موږ د “ځانګړي متغير” اصطلاح کاروو، که څه هم په
پورتنۍ قاعده کښې $a$ !!a{constant} دے۔ دا نوم تاريخي لاملونه لري۔}

\subsection{د $\lexists$ دپاره قاعدې}

\begin{defish}
\Axiom$ !A(a), \Gamma \fCenter \Delta $
\RightLabel{\LeftR{\lexists}}
\UnaryInf$ \lexists[x][!A(x)], \Gamma \fCenter \Delta$
\DisplayProof
\hfill
\Axiom$ \Gamma \fCenter \Delta, !A(t) $
\RightLabel{\RightR{\lexists}}
\UnaryInf$ \Gamma \fCenter \Delta, \lexists[x][!A(x)]$
\DisplayProof
\end{defish}

بيا هم، $t$~يوه تړلې اصطلاح ده، او $a$~داسې !!a{constant} دے چې
د \LeftR{\lexists} قاعدې په لاندې سېکوېنټ کښې نۀ راځي۔ موږ $a$ د
\LeftR{\lexists} استنتاج \emph{ځانګړے متغير} بولو۔

دا شرط چې ځانګړے متغير د \RightR{\lforall} يا
\LeftR{\lexists} استنتاج په لاندې سېکوېنټ کښې نۀ راځي،
\emph{د ځانګړي متغير شرط} بلل کېږي۔

\begin{explain}
هغه دود راياد کړئ چې کله $!A$ داسې !!a{formula} وي چې
!!{variable}~$x$ پکې ازاد وي، نو دا په~$!A(x)$ ليکلو ښيو۔ په همدې
چوکاټ کښې بيا $!A(t)$ د~$\Subst{!A}{t}{x}$ لنډون دے۔ نو د
\RightR{\lexists} قاعده داسې هم ليکلے شو:
\begin{prooftree}
  \Axiom$\Gamma \fCenter \Delta, \Subst{!A}{t}{x}$
  \RightLabel{\RightR{\lexists}}
  \UnaryInf$\Gamma \fCenter \Delta, \lexists[x][!A]$
\end{prooftree}
پام وکړئ چې $t$ کېدے شي له مخکښې په~$!A$ کښې راغلے وي؛ مثلاً
$!A$~کېدے شي~$\Atom{\Obj P}{t,x}$ وي۔ له دې امله، له
~$\Gamma \Sequent \Delta, \Atom{\Obj P}{t,t}$ څخه
$\Gamma \Sequent \Delta, \lexists[x][\Atom{\Obj P}{t,x}]$ استنتاجول
د~\RightR{\lexists} سمه کارونه ده---تاسو د مقدمې د~$t$ يوه يا
زياتې پېښې په تړلي !!{variable}~$x$ “بدلولے” شئ، او اړينه نۀ ده
چې ټولې پېښې ئې بدلې کړئ۔ خو د \RightR{\lforall} او
~\LeftR{\lexists} د ځانګړي متغير شرطونه غواړي چې
!!{constant}~$a$ په~$!A$ کښې نۀ راځي۔ نو د~$\RightR{\lforall}$ په
کارولو سره له $\Gamma \Sequent \Delta, \Atom{\Obj P}{a,a}$ څخه
$\Gamma \Sequent \Delta, \lforall[x][\Atom{\Obj P}{a,x}]$ په سمه
توګه نۀ شئ استنتاجولے۔
\end{explain}

\begin{explain}
په \RightR{\lexists} او \LeftR{\lforall} کښې پر اصطلاح~$t$ هېڅ
بنديز نشته۔ بل لوري ته، د \LeftR{\lexists} او
\RightR{\lforall} په قاعدو کښې د ځانګړي متغير شرط غواړي چې
!!{constant}~$a$ په پورتني سېکوېنټ کښې د~$!A(a)$ څخه بهر هېڅ ځاے
نۀ راځي۔ دا شرط د نظام د سموالي د يقيني کولو دپاره اړين دے؛ يعنې
نظام يوازې معتبر سېکوېنټونه !!{derive}s کړي۔ له دې شرط پرته لاندې
استنتاجونه روا کېدل:
\begin{prooftree}
  \Axiom$!A(a) \fCenter !A(a)$
  \RightLabel{*\LeftR{\lexists}}
  \UnaryInf$\lexists[x][!A(x)] \fCenter !A(a)$
  \RightLabel{\RightR{\lforall}}
  \UnaryInf$\lexists[x][!A(x)] \fCenter \lforall[x][!A(x)]$
  \DisplayProof\bottomAlignProof
  \qquad
  \Axiom$!A(a) \fCenter !A(a)$
  \RightLabel{*\RightR{\lforall}}
  \UnaryInf$!A(a) \fCenter \lforall[x][!A(x)]$
  \RightLabel{\LeftR{\lexists}}
  \UnaryInf$\lexists[x][!A(x)] \fCenter \lforall[x][!A(x)]$
\end{prooftree}
خو $\lexists[x][!A(x)] \Sequent \lforall[x][!A(x)]$ معتبر نۀ دے۔
\end{explain}
