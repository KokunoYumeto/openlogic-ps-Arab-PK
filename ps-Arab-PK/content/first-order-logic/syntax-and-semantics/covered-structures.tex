\olsection{پوښل شوي \printtoken{P}{structure} د لومړۍ درجې ژبو دپاره}

\begin{explain}
په ياد ولرئ چې يو ترم هله \emph{تړلے} دے چې هېڅ !!{variable}s ونۀ لري۔
\end{explain}

\begin{defn}[د تړلو ترمونو !!^{value}]
که $t$ د ژبې~$\Lang L$ يو تړلے ترم او $\Struct M$ د~$\Lang L$ يو
!!{structure} وي، \emph{!!{value}}~$\Value{t}{M}$ ئې په لاندې ډول
تعريفېږي:
\begin{enumerate}
\item که $t$ يوازې !!{constant}~$c$ وي، نو
  $\Value{c}{M} = \Assign{c}{M}$۔
\item که $t$ د $\Atom{f}{t_1, \ldots, t_n}$ بڼه ولري، نو
  \[
  \Value{t}{M} = \Assign{f}{M}(\Value{t_1}{M}, \ldots,
  \Value{t_n}{M}).
  \]
\end{enumerate}
\end{defn}

\begin{defn}[پوښل شوے !!{structure}]
يو !!{structure} هله \emph{پوښل شوے} دے چې د ساحې هر غړے د کوم تړلي
ترم !!{value} وي۔
\end{defn}

\begin{ex}
فرض کړئ~$\Lang L$ هغه ژبه ده چې !!{constant}s $\Obj{zero}$،
$\Obj{one}$، $\Obj{two}$، \dots، دوه ځايه !!{predicate}~$<$، او
دوه ځايه !!{function}s $+$ او $\times$ لري۔ د~$\Lang L$
!!a{structure}~$\Struct M$ دې هغه وي چې ساحه ئې
$\Domain M = \{0, 1, 2, \ldots \}$ او ګومارنې ئې
$\Assign{\Obj{zero}}{M} = 0$، $\Assign{\Obj{one}}{M} = 1$،
$\Assign{\Obj{two}}{M} = 2$، او همداسې نورې دي۔ د دوه ځايه اړيکې د
سمبول $<$ دپاره، سټ $\Assign{<}{M}$ د ټولو هغو جوړو
$\tuple{c_1, c_2} \in \Domain{M}^2$ سټ دے چې $c_1$ له~$c_2$ څخه
کوچنے وي: مثلاً $\tuple{1, 3} \in \Assign{<}{M}$، خو
$\tuple{2, 2} \notin \Assign{<}{M}$۔ د دوه ځايه !!{function} $+$
دپاره $\Assign{+}{M}$ په معمول ډول تعريف کړئ---مثلاً
$\Assign{+}{M}(2,3)$ عدد~$5$ ته لېږي؛ د دوه ځايه
!!{function}~$\times$ دپاره هم همداسې وکړئ۔ نو د $\Obj{four}$
!!{value} پخپله~$4$ دے، او د $\times(\Obj{two},
+(\Obj{three},\Obj{zero}))$ !!{value} (يا په منځ‌ليکي نښه کښې،
$\Obj{two} \times (\Obj{three} + \Obj{zero})$) دا دے:
\begin{multline*}
\Value{\times(\Obj{two}, +(\Obj{three},\Obj{zero}))}{M} \\
\begin{aligned}
& =\Assign{\times}{M}(\Value{\Obj{two}}{M}, \Value{+(\Obj{three}, \Obj{zero})}{M})\\
& = \Assign{\times}{M}(\Value{\Obj{two}}{M}, \Assign{+}{M}(\Value{\Obj{three}}{M},
\Value{\Obj{zero}}{M})) \\
& = \Assign{\times}{M}(\Assign{\Obj{two}}{M}, \Assign{+}{M}(\Assign{\Obj{three}}{M},
\Assign{\Obj{zero}}{M})) \\
& = \Assign{\times}{M}(2, \Assign{+}{M}(3, 0)) \\
& = \Assign{\times}{M}(2, 3) \\
& = 6
\end{aligned}
\end{multline*}
\textbf{د ژباړې د نسخې سمون (OLFOL-013):} د سرچينې د محاسبې په
لومړۍ کرښه کښې د برابرۍ نښه د بلې کرښې له پيل سره تکرار شوې؛ دلته
لومړۍ تکراري نښه لرې شوې ده۔
\end{ex}

\begin{prob}
ايا $\Struct N$، يعنې د حساب معياري مدل، پوښل شوے دے؟ څرګنده ئې کړئ۔
\end{prob}
