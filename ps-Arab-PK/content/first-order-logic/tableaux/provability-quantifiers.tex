\olsection{\usetoken{S}{derivability} او سورونه}

\begin{explain}
  د بشپړتيا قضيه دا هم غواړي چې د تابلو قاعدې د~$\Proves$ په اړه
  هغه حقيقتونه ورکړي چې په دې برخه کښې ثابتېږي۔
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} که $c$ داسې ثابت وي چې په
$\Gamma$ يا~$!A(x)$ کښې نۀ راځي، او $\Gamma \Proves !A(c)$ وي،
نو $\Gamma \Proves \lforall[x][!A(x)]$۔
\end{thm}

\begin{proof}
فرض کړئ $\Gamma \Proves !A(c)$، يعنې داسې
$!B_1$، \dots، $!B_n \in \Gamma$ او د لاندې دپاره تړلے
!!{tableau} شته:
\begin{align*}
\{ \sFmla{\False}{!A(c)}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\intertext{بايد وښيو چې د لاندې دپاره هم تړلے !!{tableau} شته:}
\{ \sFmla{\False}{\lforall[x][!A(x)]}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\end{align*}
تړلے !!{tableau} واخلئ، لومړے فرض ئې په
$\sFmla{\False}{\lforall[x][!A(x)]}$ بدل کړئ، او د فرضونو وروسته
$\sFmla{\False}{!A(c)}$ ورننباسئ۔
\begin{center}
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\formula{A}(c)}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
      [\sFmla{\True}{\formula{B}_n}, 
        [][][]]]]]
\end{tableau}\qquad
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\lforall[x][\formula{A}(x)]}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
        [\sFmla{\True}{\formula{B}_n},
          [\sFmla{\False}{\formula{A}(c)}
        [][][]]]]]]
\end{tableau}
\end{center}
تابلو لا هم تړلے دے؛ ځکه ټولې هغه !!{sentence}s چې مخکښې د فرضونو
په توګه موجودې وې، اوس هم د~!!{tableau} په سر کښې موجودې دي۔
ننويستل شوې کرښه د~$\TRule{\False}{\lforall}$ قاعدې د سمې کارونې
پايله ده؛ ځکه !!{constant}~$c$ په~$!B_1$، \dots، $!B_n$ يا
$\lforall[x][!A(x)]$ کښې نۀ راځي، يعنې په نوي !!{tableau} کښې د
ننويستل شوې کرښې پاس نۀ راځي۔
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A(t) \Proves \lexists[x][!A(x)]$.}{}
\tagitem{prvAll}{$\lforall[x][!A(x)] \Proves !A(t)$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{د
  $\sFmla{\False}{\lexists[x][!A(x)]}, \sFmla{\True}{!A(t)}$ دپاره
  تړلے !!{tableau} دا دے:
  \begin{oltableau}
    [\sFmla{\False}{\lexists[x][\formula{A}(x)]}, just = \TAss
      [\sFmla{\True}{\formula{A}(t)}, just = \TAss
        [\sFmla{\False}{\formula{A}(t)}, just = {\TRule{\False}{\lexists}[1]}, close]]]
    \end{oltableau}}{}
\tagitem{prvAll}{د
  $\sFmla{\False}{\formula{A}(t)}, \sFmla{\True}{\lforall[x][!A(x)]}, $ دپاره
  تړلے !!{tableau} دا دے:
  \begin{oltableau}
    [\sFmla{\False}{\formula{A}(t)}, just = \TAss
      [\sFmla{\True}{\lforall[x][\formula{A}(x)]}, just = \TAss
        [\sFmla{\True}{\formula{A}(t)}, just = {\TRule{\True}{\lforall}[2]}, close]]]
    \end{oltableau}}{}
\end{tagenumerate}
\end{proof}
