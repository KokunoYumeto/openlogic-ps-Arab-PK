\olsection{د متناهي شاهد قضيه}

د بشپړتيا قضيې يوه مهمه پايله د متناهي شاهد قضيه ده۔ د متناهي
شاهد قضيه وايي چې که د~!!{sentence}s د يوه سټ هر
\emph{متناهي} فرعي سټ د صدق وړ وي، نو ټول سټ د صدق وړ دے، حتٰی
که سټ پخپله نامتناهي وي۔ دا هېڅ څرګنده خبره نۀ ده۔ لږ تر لږه په
لومړي نظر داسې هېڅ څۀ نۀ ښکاري چې د~!!{sentence}s د داسې نامتناهي
سټونو امکان رد کړي چې متناقض وي، خو تناقض ئې، په اصطلاح، يوازې له
نامتناهي شمېر څخه راولاړېږي۔ د متناهي شاهد قضيه وايي چې داسې حالت
ردېدلے شي: د~!!{sentence}s داسې د صدق وړ نۀ نامتناهي سټونه نشته
چې هر متناهي فرعي سټ ئې د صدق وړ وي۔ د بشپړتيا قضيې په څېر، د دې
يوه بڼه هم له استلزام سره تړاو لري: که د~!!{sentence}s يو نامتناهي
سټ کوم څۀ لازموي، نو لا له وړاندې ئې يو متناهي فرعي سټ لازموي۔

\begin{defn}
  د~!!{formula}s يو سټ~$\Gamma$ هله او يوازې هله
  \emph{په متناهي ډول د صدق وړ} دے چې هر متناهي
  $\Gamma_0 \subseteq \Gamma$ د صدق وړ وي۔
\end{defn}

\begin{thm}[د متناهي شاهد قضيه]
\ollabel{thm:compactness}
د هرې د جملو ټولګې~$\Gamma$ او هرې جملې~$!A$ دپاره لاندې خبرې سمې دي:
\begin{enumerate}
  \item $\Gamma \Entails !A$ هله او يوازې هله چې يو متناهي
    $\Gamma_0 \subseteq \Gamma$ وي، داسې چې
    $\Gamma_0 \Entails !A$۔
  \item $\Gamma$ هله او يوازې هله د صدق وړ دے چې په متناهي ډول
    د صدق وړ وي۔
\end{enumerate}
\end{thm}

\begin{proof}
موږ (2) ثابتوو۔ که~$\Gamma$ د صدق وړ وي، نو داسې
\iftag{FOL}{!!a{structure}~$\Struct{M}$}{!!a{valuation}~$\pAssign{v}$}
شته چې د ټولو~$!A \in \Gamma$ دپاره
$\iftag{FOL}{\Sat{M}}{\pSat{v}}{!A}$۔ څرګنده ده چې همدا
$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$ د~$\Gamma$ هر متناهي فرعي
سټ هم پوره کوي؛ نو~$\Gamma$ په متناهي ډول د صدق وړ دے۔

اوس فرض کړئ~$\Gamma$ په متناهي ډول د صدق وړ دے۔ بيا هر متناهي
فرعي سټ~$\Gamma_0 \subseteq \Gamma$ د صدق وړ دے۔ د سموالي له مخې
(\iftag{FOL}{%
  \tagrefs{prfAX/{fol:axd:sou:cor:consistency-soundness},
    prfSC/{fol:seq:sou:cor:consistency-soundness},
    prfND/{fol:ntd:sou:cor:consistency-soundness},
    prfTab/{fol:tab:sou:cor:consistency-soundness}}}{%
  \tagrefs{prfAX/{pl:axd:sou:cor:consistency-soundness},
    prfSC/{pl:seq:sou:cor:consistency-soundness},
    prfND/{pl:ntd:sou:cor:consistency-soundness},
    prfTab/{pl:tab:sou:cor:consistency-soundness}}})،
هر متناهي فرعي سټ سازګار دے۔ بيا د لاندې پايلې له مخې~$\Gamma$
پخپله هم بايد سازګار وي:
\iftag{FOL}{%
  \tagrefs{prfAX/{fol:axd:ptn:prop:proves-compact},
    prfSC/{fol:seq:ptn:prop:proves-compact},
    prfND/{fol:ntd:ptn:prop:proves-compact},
    prfTab/{fol:tab:ptn:prop:proves-compact}}}{%
  \tagrefs{prfAX/{pl:axd:ptn:prop:proves-compact},
    prfSC/{pl:seq:ptn:prop:proves-compact},
    prfND/{pl:ntd:ptn:prop:proves-compact},
    prfTab/{pl:tab:ptn:prop:proves-compact}}}.
د بشپړتيا له مخې (\olref[cth]{thm:completeness})، ځکه~$\Gamma$
سازګار دے، نو د صدق وړ دے۔
\end{proof}

\tagprob{FOL}
\begin{prob}
د~\olref[fol][com][com]{thm:compactness} برخه (1) ثابته کړئ۔
\end{prob}
\tagendprob

\end{prob}

د~$\Domain{M}$ يو !!a{element} ټاکي۔ آيا دا هم تضمينولے شو چې د
$\Domain{M}$ هر !!{element} کوم ترم ټاکلے وي؟ په بله وينا، آيا داسې
تيورۍ~$\Gamma$ شته چې ټول مدلونه ئې پوښلي وي؟ د متناهي شاهد قضيه
ښيي چې که~$\Gamma$ نامتناهي مدلونه ولري، داسې نۀ ده۔ دا داسې
ليدلے شو: فرض کړئ~$\Struct{M}$ د~$\Gamma$ يو نامتناهي مدل دے، او
$c$ داسې يو !!a{constant} دے چې د~$\Gamma$ په ژبه کښې نشته۔ فرض
کړئ~$\Delta$ د ټولو هغو جملو~$\eq/[c][t]$ سټ دے چې~$t$ د~$\Gamma$
په ژبه~$\Lang{L}$ کښې يو تړلے ترم وي؛ يعنې
  \[
  \Delta = \Setabs{\eq/[c][t]}{t \in \Trm[L]}.
  \]
د~$\Gamma \cup \Delta$ يو متناهي فرعي سټ د
$\Gamma' \cup \Delta'$ په بڼه ليکل کېدے شي، چې
$\Gamma' \subseteq \Gamma$ او~$\Delta' \subseteq \Delta$ وي۔ ځکه
$\Delta'$ متناهي دے، يوازې متناهي ډېر ترمونه لرلے شي۔ داسې
$a \in \Domain{M}$ واخلئ چې د~$\Domain{M}$ يو !!a{element} وي او له
هغوى څخه ئې هېڅ يو نۀ ټاکي؛ او~$\Struct{M'}$ داسې !!{structure}
واخلئ چې کټ مټ د~$\Struct{M}$ په څېر وي، خو
$\Assign{c}{M'} = a$ هم ولري۔ ځکه په~$\Delta'$ کښې د راغلو ټولو
$t$ دپاره~$a \neq \Value{t}{M}$، نو~$\Sat{M'}{\Delta'}$۔ ځکه
$\Sat{M}{\Gamma}$،~$\Gamma' \subseteq \Gamma$، او~$c$ په~$\Gamma$
کښې نۀ راځي، نو~$\Sat{M'}{\Gamma'}$ هم۔ دواړه سره يوځاے راکوي چې
د~$\Gamma \cup \Delta$ د هر متناهي فرعي سټ
$\Gamma' \cup \Delta'$ دپاره~$\Sat{M'}{\Gamma' \cup \Delta'}$۔
نو د~$\Gamma \cup \Delta$ هر متناهي فرعي سټ د صدق وړ دے۔ د
متناهي شاهد قضيې له مخې،~$\Gamma \cup \Delta$ پخپله د صدق وړ دے۔
نو داسې مدلونه شته چې~$\Sat{M}{\Gamma \cup \Delta}$۔ هر داسې
$\Struct{M}$ د~$\Gamma$ مدل دے، خو پوښلے نۀ دے، ځکه د~$\Lang{L}$
د ټولو ترمونو~$t$ دپاره
$\Value{c}{M} \neq \Value{t}{M}$۔
\end{ex}

!!{structure}~$\Struct{Q}$ کښې رښتونې دي، چې د تعريف ساحه ئې
$\Rat$ او تفسيرونه ئې څرګند دي۔~$\Gamma$ د~$\Lang{L}$ د ټولو هغو
!!{sentence}s سټ دے چې د ناطقو عددونو په اړه رښتونې دي۔ څرګنده ده
چې په~$\Rat$ (او حتٰی په~$\Real$) کښې داسې عددونه~$r$ نشته چې

کوچني وي۔ که داسې عدد موجود واے، يو \emph{بې حده کوچنے عدد} به

عددونه پکښې موجود دي۔ زموږ په ژبه کښې د تقسيم !!a{function} نشته

$\Delta$ داسې وټاکئ:
\[
\{\Obj{0} < c\}
\cup \Setabs{ c \times \num k < \Obj{1} }{k \in \PosInt}

$\Assign{c}{Q'} = 1/K$ ولري، نو د هر متناهي
