\olsection{د ثبوت لنډيز}

د بشپړتيا د قضيې ثبوت يو څه پېچلے دے، او په لومړي لوست کښې ترې
لار ورکېدل اسانه دي۔ نو راځئ د ثبوت لنډيز وړاندې کړو۔ لومړے ګام د
نظر د زاويې داسې بدلون دے چې د ثبوت لاره راښيي۔ که بشپړتيا داسې
وګڼو چې «هر کله $\Gamma \Entails !A$ وي، نو
$\Gamma \Proves !A$ وي»، ښايي د کومې مفکورې موندل هم ګران وي:
ځکه د~$\Gamma \Proves !A$ د ښودلو دپاره بايد !!a{derivation}
ومومو، او داسې نۀ ښکاري چې فرضيه~$\Gamma \Entails !A$ په دې کښې
څه مرسته راسره کوي۔ د ځينو ثبوتي نظامونو دپاره !!a{derivation}
نېغ جوړېدے شي، خو موږ لږه بله لار نيسو۔ د نظر اړين بدلون دا دے:
بشپړتيا داسې هم ويل کېدے شي چې «که~$\Gamma$ سازګار وي، نو د صدق
وړ دے»۔ ښايي د~$\Gamma$ معلومات له دې فرضيې سره يو ځاے وکاروو چې
هغه سازګار دے، او داسې \iftag{FOL}{!!a{structure}}{!!a{valuation}}
جوړ کړو چې په~$\Gamma$ کښې هر
\iftag{FOL}{!!{sentence}}{!!{formula}} پوره کړي۔ بالاخره موږ پوهېږو
چې د کوم ډول \iftag{FOL}{!!{structure}}{!!{valuation}} په لټه کښې
يو: هغه چې هماغسې وي لکه~$\Gamma$ چې ئې بيانوي!{}

که~$\Gamma$ يوازې \iftag{FOL}{اتومي
  !!{sentence}s}{!!{propositional variable}s} ولري، نو د هغې دپاره
مدل جوړول اسانه دي۔\iftag{FOL}{ فرض کړئ ټولې اتومي !!{sentence}s
  د~$\Atom{P}{a_1,\dots,a_n}$ بڼه لري، چې $a_i$ ګانې
  !!{constant}s دي۔}{} يوازې دومره پکار دي چې
\iftag{FOL}{%
  !!a{domain}~$\Domain{M}$ او د~$P$ دپاره داسې ګومارنه پيدا کړو چې
  $\Sat{M}{\Atom{P}{a_1,\ldots,a_n}}$ وي۔ دا ډېره ګرانه نۀ ده:
  $\Domain{M} = \Nat$، $\Assign{\Obj c_i}{M} = i$ وټاکئ، او د هر
  $\Atom{P}{a_1,\ldots,a_n} \in \Gamma$ دپاره جوړه
  $\tuple{k_1, \dots, k_n}$ په~$\Assign{P}{M}$ کښې واچوئ، چې
  $k_i$ د ثابت سمبول~$a_i$ شمېر دے (يعنې
  $a_i \ident \Obj c_{k_i}$)۔}{%
  داسې !!a{valuation}~$\pAssign{v}$ چې د هر $p \in \Gamma$ دپاره
  $\pSat{v}{p}$ وي۔ بس، $\pAssign{v}(p) = \True$ هله او يوازې هله
  وټاکئ چې $p \in \Gamma$ وي۔}

اوس فرض کړئ~$\Gamma$ کوم داسې !!{formula}~$\lnot !B$ لري چې
$!B$ اتومي دے۔ ښايي اندېښنه وکړو چې د
$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$ جوړونه د~$\lnot !B$ د
رښتوني کولو له امکان سره ټکر کوي۔ خو دلته د~$\Gamma$ سازګاري
کار راځي: که $\lnot !B \in \Gamma$ وي، نو
$!B \notin \Gamma$، کنه~$\Gamma$ به ناسازګار و۔ او که
$!B \notin \Gamma$ وي، نو زموږ د
$\iftag{FOL}{\Struct{M}}{\pAssign{v}}$ د جوړونې له مخې
$\iftag{FOL}{\Sat/{M}}{\pSat/{v}}{!B}$؛ نو
$\iftag{FOL}{\Sat{M}}{\pSat{v}}{\lnot !B}$۔ تر دې ځايه هر څه سم
دي۔

که~$\Gamma$ پېچلي، نااتومي فارمولونه ولري نو څه؟ فرض کړئ
$!A \land !B$ لري۔ د دې د رښتوني کولو دپاره بايد داسې پر مخ لاړ
شو لکه دواړه~$!A$ او~$!B$ چې په~$\Gamma$ کښې وي۔ او که
$!A \lor !B \in \Gamma$ وي، بايد لږ تر لږه يو ئې رښتونے کړو؛
يعنې داسې پر مخ لاړ شو لکه يو ئې چې په~$\Gamma$ کښې وي۔

دا لاندې مفکوره راښيي: په~$\Gamma$ کښې نور !!{formula}s داسې
ورزياتوو چې (الف)~ترې جوړېدونکے سټ سازګار وساتو، (ب)~ډاډ ترلاسه
کړو چې د هرې ممکنې !!{sentence}~$!A$ دپاره يا~$!A$ په
جوړ شوي سټ کښې وي يا~$\lnot !A$، او (ج)~که هر کله
$!A \land !B$ په سټ کښې وي، دواړه~$!A$ او~$!B$ هم پکښې وي؛ که
$!A \lor !B$ په سټ کښې وي، لږ تر لږه يو له~$!A$ او~$!B$ څخه هم
پکښې وي؛ او داسې نور۔ \textbf{د ژباړې د نسخې سمون (OLCOM-001):}
سرچينې په شرط~(ب) کښې «اتومي جمله» ليکلې وه، خو همدا پاراګراف دا
شرط د بشپړ سټ تعريف بولي او ورپسې رسمي تعريف ئې د هرې جملې دپاره
ټاکي؛ نو ناسمه «اتومي» ځانګړنه لرې شوه۔ موږ دې کار ته (که اړينه
وي، تر ابده) دوام
ورکوو۔ د ټولو داسې زياتو شوو !!{formula}s سټ~$\Gamma^*$ وبولئ۔
بيا به پورته جوړونه داسې
\iftag{FOL}{!!a{structure}~$\Struct{M}$}{!!a{valuation}~$\pAssign{v}$}
راکړي چې په استقرا سره ثابتولے شو په~$\Gamma^*$ کښې ټولې جملې
پوره کوي، او ځکه چې $\Gamma \subseteq \Gamma^*$، په~$\Gamma$
کښې ټولې جملې هم پوره کوي۔ څرګندېږي چې د (الف) او~(ب) تضمين بس
دے۔ د جملو هغه سټ چې (ب) پوره کوي \emph{بشپړ} بلل کېږي۔ نو زموږ
کار به دا وي چې سازګار سټ~$\Gamma$ يو سازګار او بشپړ
سټ~$\Gamma^*$ ته وغځوو۔

\iftag{FOL}{%
په دې پلان کښې يوه ستونزه شته: که
$\lexists[x][!A(x)] \in \Gamma$ وي، غواړو په دې بهير کښې کوم
!!{constant}~$c$ وټاکو او~$!A(c)$ ورزيات کړو۔ خو څنګه پوهېږو چې
دا کار تل کېدے شي؟ ښايي زموږ په ژبه کښې يوازې څو !!{constant}s
وي، او د هر يوه دپاره $\lnot !A(c) \in \Gamma$ ولرو۔ ورسره
$!A(c)$ هم نۀ شو زياتولے، ځکه سټ به ناسازګار کړي؛ بيا به نۀ پوهېدو
چې~$\Struct{M}$ بايد~$!A(c)$ رښتونے کړي که~$\lnot !A(c)$۔ لا دا
هم کېدے شي چې~$\Gamma$ يوازې د داسې ژبې جملې ولري چې هېڅ ثابت
سمبول نۀ لري (د بېلګې په توګه، د سټونو د تيورۍ ژبه)۔

د دې ستونزې حل دا دے چې په پيل کښې بېخي نامتناهي ډېر ثابتونه، او
ورسره داسې جملې، زياتې کړو چې په سم ډول ئې له کميت ټاکونکو سره
وتړي۔ (البته بايد تصديق کړو چې دا کار ناسازګاري نۀ شي رامنځته
کولے۔)

زموږ لومړنۍ جوړونه هغه وخت ښه کار کوي چې په اتومي جملو کښې يوازې
!!{constant}s ولرو۔ خو ژبه ښايي !!{function}s هم ولري۔ په هغه
حالت کښې ښايي پر~$\Nat$ د سمې تابعې موندل، چې دغو !!{function}s
ته ئې وګومارو او هر څه سم شي، ستونزمن وي۔ نو بله چاره دا ده: د
$\Obj c_i$ د تفسير دپاره د~$i$ د کارولو پر ځاے، د !!{constant}s
خپله سټ د تعريف ساحه وټاکئ۔ بيا~$\Struct M$ هر !!{constant} خپل
ځان ته ګومارلے شي: $\Assign{\Obj c_i}{M} = \Obj c_i$۔ خو ولې دا
کار تر پايه نۀ رسوو: د ژبې ټول \emph{ترمونه}~$\Domain{M}$ وټاکئ!{}
په دې صورت کښې !!{function}s ته د تابعو څرګنده ګومارنه شته (دغه
تابعې ترمونه د ارګومېنټونو په توګه اخلي او ترمونه د قيمتونو په توګه
لري): !!{function}~$\Obj f^n_i$ ته هغه تابع ګومارو چې $n$ ترمونه
$t_1$، \dots،~$t_n$ د ننوت په توګه واخلي او ترم
$\Obj f^n_i(t_1, \dots, t_n)$ د قيمت په توګه ورکړي۔

د معما وروستۍ ټوټه دا ده چې له~$\eq$ سره څه وکړو۔
!!{predicate}~$\eq$ ثابت تفسير لري:
$\Sat{M}{\eq[t][t']}$ هله او يوازې هله چې
$\Value{t}{M} = \Value{t'}{M}$ وي۔ اوس که هر څه داسې وټاکو چې د
ترم~$t$ !!{value} په خپله~$t$ وي، نو دا !!{structure} به د
$\eq[t][t']$ په بڼه \emph{هېڅ} جمله رښتونې نۀ کړي، مګر دا چې
$t$ او~$t'$ يو او هماغه ترم وي۔ څرګنده ده چې دا ستونزه ده، ځکه د
!!{function}s لرونکې ژبې کابو هره په زړه پورې تيوري به داسې جملې
$\eq[t][t']$ د قضيو په توګه ولري چې~$t$ او~$t'$ يو ترم نۀ وي (د
بېلګې په توګه، د حساب په تيوريو کښې:
$\eq[(\Obj 0+ \Obj 0)][\Obj 0]$)۔ د دې ستونزې د هوارولو دپاره د
$\Struct M$ تعريف ساحه بدلوو: په~$\Domain{M}$ کښې د څيزونو په
توګه د ترمونو پر ځاے د ترمونو سټونه کاروو، او هر سټ ټول هغه ترمونه
لري چې د~$\Gamma$ جملې ئې برابر ګڼل غواړي۔ نو د بېلګې په توګه، که
$\Gamma$ د حساب تيوري وي، له دغو سټونو څخه يو به~$\Obj 0$،
$(\Obj 0 + \Obj 0)$، $(\Obj 0 \times \Obj 0)$، او داسې نور ولري۔
همدا سټ به~$\Obj 0$ ته ګومارو، او څرګنده به شي چې همدا سټ په دې
کښې د ټولو ترمونو قيمت هم دے، لکه د~$(\Obj 0 + \Obj 0)$ قيمت۔
نو په دې بدله شوې !!{structure} کښې به جمله
$\eq[(\Obj 0+ \Obj 0)][\Obj 0]$ رښتونې وي۔}{}

نو دا کارونه به کوو۔ لومړے د !!{complete} سازګارو سټونو خاصيتونه
څېړو؛ په ځانګړي ډول ثابتوو چې !!a{complete} سازګار سټ
$!A \land !B$ هله او يوازې هله لري چې دواړه~$!A$ او~$!B$ ولري،
او~$!A \lor !B$ هله او يوازې هله لري چې لږ تر لږه يو ئې ولري؛
او داسې نور (\olref[ccs]{prop:ccs})۔\iftag{FOL}{ بيا د جملو
  «مشبوع» سټونه تعريفوو او څېړو۔ مشبوع سټ داسې شرطي فارمولونه لري
  چې هره کميت ټاکونکې !!{sentence} د هغې له بېلګو سره تړي
  (\olref[hen]{defn:henkin-exp})۔ ښيو چې هر سازګار سټ~$\Gamma$
  تل مشبوع سټ~$\Gamma'$ ته غځېدے شي (\olref[hen]{lem:henkin})۔ که
  يو سټ سازګار، مشبوع او !!{complete} وي، دا خاصيت هم لري چې
  \iftag{prvEx}{$\lexists[x][!A(x)]$ هله او يوازې هله لري چې د کوم
    تړلي ترم~$t$ دپاره~$!A(t)$ ولري}{}\iftag{defEx,defAll}{}{ او
  }\iftag{prvAll}{$\lforall[x][!A(x)]$ هله او يوازې هله لري چې د
    ټولو تړلو ترمونو~$t$ دپاره~$!A(t)$ ولري}{}
  (\olref[hen]{prop:saturated-instances})۔}{}
بيا \iftag{FOL}{مشبوع}{} سازګار سټ
$\iftag{FOL}{\Gamma'}{\Gamma}$ اخلو، او ښيو چې داسې
\iftag{FOL}{مشبوع، سازګار او}{سازګار او} !!{complete}
سټ~$\Gamma^*$ ته غځېدے شي (\olref[lin]{lem:lindenbaum})۔ همدا
سټ~$\Gamma^*$ د خپل \iftag{FOL}{ترمي مدل~$\Struct M(\Gamma^*)$}
{!!{valuation}~$\pAssign v(\Gamma^*)$} د تعريف دپاره کاروو۔
\iftag{FOL}{د ترمي مدل تعريف ساحه د تړلو ترمونو سټ دے، او د هغه
  د !!{predicate}s تفسير د هغو اتومي
  !!{sentence}s}{ارزښت ټاکنه د هغو !!{propositional
    variable}s} له مخې ټاکل کېږي چې په~$\Gamma^*$ کښې دي
(\olref[mod]{defn:termmodel})۔ د \iftag{FOL}{مشبوع،}{} بشپړو
سازګارو سټونو خاصيتونه کاروو، څو وښيو چې په رښتيا
$\iftag{FOL}{\Sat{M(\Gamma^*)}}{\pSat{v(\Gamma^*)}}{!A}$ هله او
يوازې هله چې $!A \in \Gamma^*$
(\olref[mod]{lem:truth})؛ نو په ځانګړي ډول
$\iftag{FOL}{\Sat{M(\Gamma^*)}}{\pSat{v(\Gamma^*)}}{\Gamma}$۔
\iftag{FOL}{په پاے کښې دا هم څېړو چې که~$\Gamma$،~$\eq$ هم ولري،
  ترمي مدل څنګه تعريف کړو (\olref[ide]{defn:term-model-factor})،
  او ښيو چې هغه~$\Gamma^*$ پوره کوي (\olref[ide]{lem:truth})۔}{}
